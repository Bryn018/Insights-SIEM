  1182→function relativeTime(iso: string): string {
  1183→  const diff = Date.now() - new Date(iso).getTime()
  1184→  const mins = Math.floor(diff / 60000)
  1185→  if (mins < 60) return `${mins}m ago`
  1186→  const hours = Math.floor(mins / 60)
  1187→  if (hours < 24) return `${hours}h ago`
  1188→  const days = Math.floor(hours / 24)
  1189→  if (days < 30) return `${days}d ago`
  1190→  return new Date(iso).toLocaleDateString()
  1191→}
