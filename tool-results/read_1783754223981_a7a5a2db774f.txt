   109→function timeAgo(dateStr: string): string {
   110→  const now = Date.now()
   111→  const then = new Date(dateStr).getTime()
   112→  const diff = now - then
   113→  if (diff < 60_000) return 'just now'
   114→  if (diff < 3_600_000) return `${Math.floor(diff / 60_000)}m ago`
   115→  if (diff < 86_400_000) return `${Math.floor(diff / 3_600_000)}h ago`
   116→  return `${Math.floor(diff / 86_400_000)}d ago`
   117→}

   123→// ===== Animated Counter Hook =====
   124→
   125→function useAnimatedCounter(target: number, duration = 1200, enabled = true) {
   126→  const [value, setValue] = useState(0)
   127→  const rafRef = useRef<number>(0)
   128→  const prevTargetRef = useRef(0)
   129→
   130→  useEffect(() => {
   131→    if (!enabled) {
   132→      // Directly use target when animation is disabled
   133→      prevTargetRef.current = target
   134→      return
   135→    }
   136→
   137→    const startVal = prevTargetRef.current
   138→    const start = performance.now()
   139→    const animate = (now: number) => {
   140→      const elapsed = now - start
   141→      const progress = Math.min(elapsed / duration, 1)
   142→      // easeOutCubic
   143→      const eased = 1 - Math.pow(1 - progress, 3)
   144→      const current = Math.round(startVal + eased * (target - startVal))
   145→      setValue(current)
   146→      if (progress < 1) {
   147→        rafRef.current = requestAnimationFrame(animate)
   148→      } else {
   149→        prevTargetRef.current = target
   150→      }
   151→    }
   152→    rafRef.current = requestAnimationFrame(animate)
   153→    return () => cancelAnimationFrame(rafRef.current)
   154→  }, [target, duration, enabled])
   155→
   156→  return value
   157→}

   159→// ===== Circular Progress Component =====
   160→
   161→function CircularProgress({
   162→  value,
   163→  size = 48,
   164→  strokeWidth = 4,
   165→}: {
   166→  value: number
   167→  size?: number
   168→  strokeWidth?: number
   169→}) {
   170→  const radius = (size - strokeWidth) / 2
   171→  const circumference = 2 * Math.PI * radius
   172→  const offset = circumference - (value / 100) * circumference
   173→
   174→  const color =
   175→    value >= 80 ? '#10b981' : value >= 60 ? '#eab308' : value >= 40 ? '#f97316' : '#ef4444'
   176→
   177→  return (
   178→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   179→      <svg width={size} height={size} className="-rotate-90">
   180→        <circle
   181→          cx={size / 2}
   182→          cy={size / 2}
   183→          r={radius}
   184→          fill="none"
   185→          stroke="rgba(255,255,255,0.08)"
   186→          strokeWidth={strokeWidth}
   187→        />
   188→        <circle
   189→          cx={size / 2}
   190→          cy={size / 2}
   191→          r={radius}
   192→          fill="none"
   193→          stroke={color}
   194→          strokeWidth={strokeWidth}
   195→          strokeDasharray={circumference}
   196→          strokeDashoffset={offset}
   197→          strokeLinecap="round"
   198→          className="transition-all duration-700 ease-out"
   199→        />
   200→      </svg>
   201→      <span className="absolute text-xs font-bold" style={{ color }}>
   202→        {value}%
   203→      </span>
   204→    </div>
   205→  )

   476→// ===== Mini Sparkline Component =====
   477→
   478→function MiniSparkline({
   479→  data,
   480→  color = '#10b981',
   481→  height = 28,
   482→  width = 60,
   483→}: {
   484→  data: number[]
   485→  color?: string
   486→  height?: number
   487→  width?: number
   488→}) {
   489→  if (data.length < 2) return null
   490→
   491→  const max = Math.max(...data, 1)
   492→  const min = Math.min(...data, 0)
   493→  const range = max - min || 1
   494→
   495→  const points = data
   496→    .map((v, i) => {
   497→      const x = (i / (data.length - 1)) * width
   498→      const y = height - ((v - min) / range) * (height - 4) - 2
   499→      return `${x},${y}`
   500→    })
   501→    .join(' ')
   502→
   503→  return (
   504→    <svg width={width} height={height} className="opacity-60">
   505→      <polyline fill="none" stroke={color} strokeWidth={1.5} points={points} />
   506→    </svg>
   507→  )
   508→}
