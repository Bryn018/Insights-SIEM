    59→// ===== Tiny deterministic sparkline =====
    60→// Generates a stable 8-point trend from the rule id + hitCount so the
    61→// chart doesn't reshuffle on every render. No new deps required.
    62→
    63→function hashString(s: string): number {
    64→  let h = 2166136261
    65→  for (let i = 0; i < s.length; i++) {
    66→    h ^= s.charCodeAt(i)
    67→    h = (h * 16777619) >>> 0
    68→  }
    69→  return h
    70→}
    71→
    72→function generateSparklinePoints(ruleId: string, hitCount: number): number[] {
    73→  if (hitCount <= 0) return [0, 0, 0, 0, 0, 0, 0, 0]
    74→  const seed = hashString(ruleId)
    75→  const points: number[] = []
    76→  let prev = hitCount * 0.4
    77→  for (let i = 0; i < 8; i++) {
    78→    // pseudo-random walk derived from the seed
    79→    const r = ((seed >> (i * 3)) & 0xff) / 255 - 0.45
    80→    prev = Math.max(0, prev + r * hitCount * 0.25)
    81→    points.push(prev)
    82→  }
    83→  // Force the last point to the actual hitCount so the trend "lands" on reality
    84→  points[points.length - 1] = hitCount
    85→  return points
    86→}
    87→
    88→function HitsSparkline({ ruleId, hitCount }: { ruleId: string; hitCount: number }) {
    89→  const points = useMemo(
    90→    () => generateSparklinePoints(ruleId, hitCount),
    91→    [ruleId, hitCount]
    92→  )
    93→  const max = Math.max(1, ...points)
    94→  const w = 44
    95→  const h = 14
    96→  const stepX = w / (points.length - 1)
    97→  const path = points
    98→    .map((p, i) => {
    99→      const x = i * stepX
   100→      const y = h - (p / max) * (h - 2) - 1
   101→      return `${i === 0 ? 'M' : 'L'}${x.toFixed(1)},${y.toFixed(1)}`
   102→    })
   103→    .join(' ')
   104→  const areaPath = `${path} L${w},${h} L0,${h} Z`
   105→  const color =
   106→    hitCount > 100 ? '#ef4444' : hitCount > 20 ? '#f97316' : hitCount > 0 ? '#10b981' : '#52525b'
   107→
   108→  if (hitCount === 0) {
   109→    return (
   110→      <svg width={w} height={h} className="shrink-0 opacity-50" aria-hidden="true">
   111→        <line x1="0" y1={h - 1} x2={w} y2={h - 1} stroke={color} strokeWidth="1" strokeDasharray="2 2" />
   112→      </svg>
   113→    )
   114→  }
   115→
   116→  return (
   117→    <svg width={w} height={h} className="shrink-0" aria-hidden="true">
   118→      <defs>
   119→        <linearGradient id={`spark-${ruleId}`} x1="0" y1="0" x2="0" y2="1">
   120→          <stop offset="0%" stopColor={color} stopOpacity="0.35" />
   121→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   122→        </linearGradient>
   123→      </defs>
   124→      <path d={areaPath} fill={`url(#spark-${ruleId})`} />
   125→      <path d={path} fill="none" stroke={color} strokeWidth="1.25" strokeLinecap="round" strokeLinejoin="round" />
   126→    </svg>
   127→  )
   128→}

   156→
   157→function getMitreTooltipText(tactic: string | null, technique: string | null): string {
   158→  if (technique) {
   159→    // Normalise T1059.001 -> T1059_001 for lookup
   160→    const key = technique.replace('.', '_')
   161→    const desc = MITRE_TECHNIQUE_DESCRIPTIONS[key]
   162→    if (desc) return `${technique}: ${desc}`
   163→  }
   164→  if (tactic && technique) return `Tactic: ${tactic}\nTechnique: ${technique}`
   165→  if (tactic) return `Tactic: ${tactic}`
   166→  if (technique) return `Technique: ${technique}`
   167→  return 'No MITRE ATT&CK mapping available'
   168→}
