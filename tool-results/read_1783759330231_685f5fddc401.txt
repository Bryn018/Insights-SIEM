   190→// ===== 10-3: CountUp helper — animates from previous value to new value =====
   191→function CountUp({ value, className }: { value: number; className?: string }) {
   192→  const [display, setDisplay] = useState(value)
   193→  const fromRef = useRef(value)
   194→  const rafRef = useRef<number | null>(null)
   195→
   196→  useEffect(() => {
   197→    const from = fromRef.current
   198→    const to = value
   199→    if (from === to) return
   200→    const duration = 600
   201→    const startTs = performance.now()
   202→    const tick = (now: number) => {
   203→      const t = Math.min(1, (now - startTs) / duration)
   204→      // easeOutCubic
   205→      const eased = 1 - Math.pow(1 - t, 3)
   206→      setDisplay(Math.round(from + (to - from) * eased))
   207→      if (t < 1) {
   208→        rafRef.current = requestAnimationFrame(tick)
   209→      } else {
   210→        fromRef.current = to
   211→        rafRef.current = null
   212→      }
   213→    }
   214→    rafRef.current = requestAnimationFrame(tick)
   215→    return () => {
   216→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current)
   217→      fromRef.current = value
   218→    }
   219→  }, [value])
   220→
   221→  return <span className={className}>{display.toLocaleString()}</span>
   222→}

  1548→            onFetchDetail()

  1554→        try {
  1555→          const res = await fetch(`/api/alerts/${alert.id}/assign`, {
  1556→            method: 'POST',
  1557→            headers: { 'Content-Type': 'application/json' },

  1561→            toast.success('Alert assigned')
  1562→            onRefresh()
  1563→            onFetchDetail()
  1564→          }
  1565→        } catch {
  1566→          toast.error('Failed to assign alert')
  1567→        }
  1568→      } else if (action === 'link_incident') {
  1569→        useSIEMStore.getState().setCreateIncidentOpen(true)
  1570→      } else if (action === 'view_details') {
  1571→        onOpenDrawer()
  1572→      }
  1573→    },
  1574→    [alert.id, onStatusChange, onRefresh, onFetchDetail, onExpand, onOpenDrawer]
  1575→  )
  1576→
  1577→  const tags = useMemo(() => {
  1578→    if (!alert.tags) return []
  1579→    try {
  1580→      return JSON.parse(alert.tags)

  1582→      return alert.tags.split(',').map((t: string) => t.trim()).filter(Boolean)
