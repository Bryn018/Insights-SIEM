   340→// ===== Helper: Parse JSON setting value =====
   341→
   342→function parseSettingValue(value: string | null | undefined, fallback: string = ''): string {
   343→  if (!value) return fallback
   344→  try {
   345→    const parsed = JSON.parse(value)
   346→    if (typeof parsed === 'object' && parsed !== null && 'value' in parsed) {
   347→      return String((parsed as { value: unknown }).value)
   348→    }
   349→    return String(parsed)
   350→  } catch {
   351→    return value
   352→  }
   353→}

   355→// ===== Helper: Format Events/Day =====
   356→
   357→function formatEventsPerDay(count: number): string {
   358→  if (count >= 1000000) return `${(count / 1000000).toFixed(1)}M`
   359→  if (count >= 1000) return `${(count / 1000).toFixed(0)}K`
   360→  return count.toString()
   361→}
