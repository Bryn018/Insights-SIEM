    97→// ===== Types =====
    98→
    99→type CasePriority = 'P1' | 'P2' | 'P3' | 'P4'
   100→type CaseStatus = 'Open' | 'In Progress' | 'Pending Review' | 'Closed'
   101→type CaseCategory =
   102→  | 'Malware'
   103→  | 'Phishing'
   104→  | 'Insider Threat'
   105→  | 'Data Breach'
   106→  | 'Policy Violation'
   107→  | 'Other'
   108→
   109→type EvidenceType = 'File' | 'Image' | 'Log' | 'Screenshot' | 'Network Capture'
   110→type TimelineEventType =
   111→  | 'Created'
   112→  | 'Assigned'
   113→  | 'Status Changed'
   114→  | 'Comment Added'
   115→  | 'Evidence Added'
   116→  | 'Priority Changed'
   117→  | 'Linked Item'
   118→  | 'Closed'
   119→
   120→interface Evidence {
   121→  id: string
   122→  type: EvidenceType
   123→  filename: string
   124→  size: string
   125→  uploadedBy: string
   126→  uploadedAt: string
   127→  description: string
   128→}
   129→
   130→interface CaseComment {
   131→  id: string
   132→  author: string
   133→  role: string
   134→  initials: string
   135→  color: string
   136→  text: string
   137→  createdAt: string
   138→}
   139→
   140→interface TimelineEvent {
   141→  id: string
   142→  type: TimelineEventType
   143→  user: string
   144→  description: string
   145→  timestamp: string
   146→}
   147→
   148→interface LinkedItem {
   149→  id: string
   150→  type: 'Alert' | 'Incident' | 'IOC'
   151→  label: string
   152→  severity: 'critical' | 'high' | 'medium' | 'low'
   153→}
   154→
   155→interface CaseItem {
   156→  id: string
   157→  title: string
   158→  description: string
   159→  priority: CasePriority
   160→  status: CaseStatus
   161→  category: CaseCategory
   162→  assignee: { name: string; initials: string; color: string }
   163→  tags: string[]
   164→  createdAt: string
   165→  updatedAt: string
   166→  resolutionTime?: string
   167→  evidence: Evidence[]
   168→  comments: CaseComment[]
   169→  timeline: TimelineEvent[]
   170→  linkedItems: LinkedItem[]
   171→}

   738→function formatRelativeTime(iso: string): string {
   739→  const date = new Date(iso)
   740→  const now = new Date()
   741→  const diffMs = now.getTime() - date.getTime()
   742→  const diffMin = Math.floor(diffMs / 60000)
   743→  const diffHr = Math.floor(diffMin / 60)
   744→  const diffDay = Math.floor(diffHr / 24)
   745→
   746→  if (diffMin < 1) return 'just now'
   747→  if (diffMin < 60) return `${diffMin}m ago`
   748→  if (diffHr < 24) return `${diffHr}h ago`
   749→  if (diffDay < 30) return `${diffDay}d ago`
   750→  return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   751→}

   916→function DonutChart({ data }: { data: Array<{ label: string; count: number; color: string }> }) {
   917→  const total = data.reduce((s, d) => s + d.count, 0) || 1
   918→  const radius = 56
   919→  const stroke = 14
   920→  const circumference = 2 * Math.PI * radius
   921→
   922→  // Precompute segment offsets using reduce (no reassignment during render)
   923→  const segments = data.reduce<Array<{ color: string; dash: number; offset: number }>>(
   924→    (acc, d) => {
   925→      const dash = (d.count / total) * circumference
   926→      const offset = acc.length > 0 ? acc[acc.length - 1].offset + acc[acc.length - 1].dash : 0
   927→      return [...acc, { color: d.color, dash, offset }]
   928→    },
   929→    []
   930→  )
   931→
   932→  return (
   933→    <div className="flex items-center gap-4">
   934→      <svg width="140" height="140" viewBox="0 0 140 140" className="shrink-0">
   935→        <circle cx="70" cy="70" r={radius} fill="none" stroke="#27272a" strokeWidth={stroke} />
   936→        {segments.map((seg, i) => (
   937→          <motion.circle
   938→            key={i}
   939→            cx="70"
   940→            cy="70"
   941→            r={radius}
   942→            fill="none"
   943→            stroke={seg.color}
   944→            strokeWidth={stroke}
   945→            strokeDasharray={`${seg.dash} ${circumference - seg.dash}`}
   946→            strokeDashoffset={-seg.offset}
   947→            transform="rotate(-90 70 70)"
   948→            initial={{ opacity: 0 }}
   949→            animate={{ opacity: 1 }}
   950→            transition={{ delay: 0.1 * i }}
   951→          />
   952→        ))}
   953→        <text
   954→          x="70"
   955→          y="65"
   956→          textAnchor="middle"
   957→          className="fill-zinc-100 font-bold"
   958→          style={{ fontSize: 22 }}
   959→        >
   960→          {total}
   961→        </text>
   962→        <text
   963→          x="70"
   964→          y="85"
   965→          textAnchor="middle"
   966→          className="fill-zinc-500"
   967→          style={{ fontSize: 10, letterSpacing: 1 }}
   968→        >
   969→          CASES
   970→        </text>
   971→      </svg>
   972→      <div className="space-y-1.5">
   973→        {data.map((d, i) => (
   974→          <div key={i} className="flex items-center gap-2 text-xs">
   975→            <span className="size-2.5 rounded-sm" style={{ background: d.color }} />

   985→function Sparkline({ data, color = '#10b981' }: { data: number[]; color?: string }) {
   986→  const width = 280
   987→  const height = 50
   988→  const max = Math.max(...data, 1)
   989→  const min = Math.min(...data, 0)
   990→  const range = max - min || 1
   991→  const step = width / (data.length - 1 || 1)
   992→
   993→  const points = data
   994→    .map((v, i) => `${i * step},${height - ((v - min) / range) * (height - 6) - 3}`)
   995→    .join(' ')
   996→
   997→  return (
   998→    <svg width={width} height={height} className="w-full">
   999→      <defs>
  1000→        <linearGradient id="spark-fill" x1="0" y1="0" x2="0" y2="1">
  1001→          <stop offset="0%" stopColor={color} stopOpacity="0.3" />
  1002→          <stop offset="100%" stopColor={color} stopOpacity="0" />
  1003→        </linearGradient>
  1004→      </defs>
  1005→      <polyline
  1006→        points={`0,${height} ${points} ${width},${height}`}
  1007→        fill="url(#spark-fill)"
  1008→        stroke="none"
  1009→      />
  1010→      <motion.polyline
  1011→        points={points}
  1012→        fill="none"
  1013→        stroke={color}
  1014→        strokeWidth="2"
  1015→        strokeLinejoin="round"
  1016→        strokeLinecap="round"
  1017→        initial={{ pathLength: 0 }}
  1018→        animate={{ pathLength: 1 }}
  1019→        transition={{ duration: 1 }}
  1020→      />
  1021→    </svg>
  1022→  )
  1023→}
