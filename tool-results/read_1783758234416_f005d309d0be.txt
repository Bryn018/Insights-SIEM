   167→function AssetStatsCards({ assets, atRiskCount }: { assets: Asset[]; atRiskCount: number }) {
   168→  const stats = useMemo(() => {
   169→    const total = assets.length
   170→    const active = assets.filter((a) => a.status === 'active').length
   171→    const critical = assets.filter((a) => a.criticality === 'critical').length
   172→    return { total, active, critical, atRisk: atRiskCount }
   173→  }, [assets, atRiskCount])
   174→
   175→  const cards = [
   176→    {
   177→      label: 'Total Assets',
   178→      value: stats.total,
   179→      icon: Network,
   180→      color: 'text-zinc-200',
   181→      bg: 'bg-zinc-800/50',
   182→      border: 'border-zinc-700',
   183→    },
   184→    {
   185→      label: 'Active',
   186→      value: stats.active,
   187→      icon: Activity,
   188→      color: 'text-emerald-400',
   189→      bg: 'bg-emerald-500/5',
   190→      border: 'border-emerald-500/20',
   191→    },
   192→    {
   193→      label: 'Critical',
   194→      value: stats.critical,
   195→      icon: AlertTriangle,
   196→      color: 'text-red-400',
   197→      bg: 'bg-red-500/5',
   198→      border: 'border-red-500/20',
   199→    },
   200→    {
   201→      label: 'At Risk',
   202→      value: stats.atRisk,
   203→      icon: ShieldAlert,
   204→      color: 'text-amber-400',
   205→      bg: 'bg-amber-500/5',
   206→      border: 'border-amber-500/20',
   207→    },
   208→  ]
   209→
   210→  return (
   211→    <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   212→      {cards.map((card) => (
   213→        <Card
   214→          key={card.label}
   215→          className={cn('rounded-xl border p-4 transition-colors hover:border-opacity-60', card.bg, card.border)}
   216→        >
   217→          <div className="flex items-center justify-between">
   218→            <span className="text-[10px] font-semibold uppercase tracking-wider text-zinc-500">
   219→              {card.label}
   220→            </span>
   221→            <card.icon className={cn('h-4 w-4', card.color)} />
   222→          </div>
   223→          <div className={cn('mt-1 text-2xl font-bold tabular-nums', card.color)}>
   224→            {card.value.toLocaleString()}
   225→          </div>
   226→        </Card>
   227→      ))}
   228→    </div>
   229→  )
   230→}

   232→// ===== Filter Toggle Button =====
   233→// 10-3: enhanced with an optional "All" pill that is active when nothing is
   234→// selected and clears the filter on click.
   235→
   236→function FilterToggle({
   237→  label,
   238→  options,
   239→  selected,
   240→  onToggle,
   241→  showAll = false,
   242→}: {
   243→  label: string
   244→  options: { value: string; label: string }[]
   245→  selected: string[]
   246→  onToggle: (value: string) => void
   247→  showAll?: boolean
   248→}) {
   249→  const clearAll = useCallback(() => {
   250→    // Toggle each currently-selected value off to clear the filter
   251→    selected.forEach((v) => onToggle(v))
   252→  }, [selected, onToggle])
   253→
   254→  return (
   255→    <div className="flex flex-wrap items-center gap-1">
   256→      <span className="text-[10px] font-semibold uppercase tracking-wider text-zinc-500 mr-1">
   257→        {label}:
   258→      </span>
   259→      {showAll && (
   260→        <Button
   261→          variant="ghost"
   262→          size="sm"
   263→          className={cn(
   264→            'h-6 px-2 text-[10px] rounded-full border transition-colors',
   265→            selected.length === 0
   266→              ? 'bg-emerald-600/20 text-emerald-400 border-emerald-500/30'
   267→              : 'text-zinc-500 border-zinc-700 hover:text-zinc-300 hover:border-zinc-600'
   268→          )}
   269→          onClick={clearAll}
   270→        >
   271→          All
   272→        </Button>
   273→      )}
   274→      {options.map((opt) => {
   275→        const isActive = selected.includes(opt.value)
   276→        return (
   277→          <Button
   278→            key={opt.value}
   279→            variant="ghost"
   280→            size="sm"
   281→            className={cn(
   282→              'h-6 px-2 text-[10px] rounded-full border transition-colors',
   283→              isActive
   284→                ? 'bg-emerald-600/20 text-emerald-400 border-emerald-500/30'
   285→                : 'text-zinc-500 border-zinc-700 hover:text-zinc-300 hover:border-zinc-600'
   286→            )}
   287→            onClick={() => onToggle(opt.value)}
   288→          >
   289→            {opt.label}
   290→          </Button>
   291→        )
   292→      })}
   293→    </div>
   294→  )
   295→}

   680→function HealthRing({ status, size = 44 }: { status: AssetStatus; size?: number }) {

   685→  const pct = status === 'active' ? 100 : status === 'maintenance' ? 55 : status === 'inactive' ? 20 : 0

   688→    status === 'active' ? '#10b981' : status === 'maintenance' ? '#f59e0b' : status === 'inactive' ? '#ef4444' : '#71717a'

   692→      width={size}
   693→      height={size}

   695→      className="siem-health-ring shrink-0"
   696→      style={{
   697→        ['--siem-ring-circumference' as string]: `${circumference}`,
   698→        ['--siem-ring-offset' as string]: `${offset}`,
   699→      } as React.CSSProperties}

   701→      <circle
   702→        className="siem-health-ring-track"

   704→        cy={size / 2}
   705→        r={radius}
   706→        strokeWidth={stroke}

   709→        className="siem-health-ring-progress"
   710→        cx={size / 2}
   711→        cy={size / 2}

   714→        stroke={strokeColor}
