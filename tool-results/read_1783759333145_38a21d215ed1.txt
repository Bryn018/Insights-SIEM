   114→function timeAgo(dateStr: string): string {
   115→  const now = Date.now()
   116→  const then = new Date(dateStr).getTime()
   117→  const diff = now - then
   118→  if (diff < 60_000) return 'just now'
   119→  if (diff < 3_600_000) return `${Math.floor(diff / 60_000)}m ago`
   120→  if (diff < 86_400_000) return `${Math.floor(diff / 3_600_000)}h ago`
   121→  return `${Math.floor(diff / 86_400_000)}d ago`
   122→}

   128→// ===== Animated Counter Hook =====

   130→function useAnimatedCounter(target: number, duration = 1200, enabled = true) {
   131→  const [value, setValue] = useState(0)
   132→  const rafRef = useRef<number>(0)
   133→  const prevTargetRef = useRef(0)
   134→
   135→  useEffect(() => {
   136→    if (!enabled) {
   137→      // Directly use target when animation is disabled
   138→      prevTargetRef.current = target
   139→      return
   140→    }
   141→
   142→    const startVal = prevTargetRef.current
   143→    const start = performance.now()
   144→    const animate = (now: number) => {
   145→      const elapsed = now - start
   146→      const progress = Math.min(elapsed / duration, 1)
   147→      // easeOutCubic
   148→      const eased = 1 - Math.pow(1 - progress, 3)
   149→      const current = Math.round(startVal + eased * (target - startVal))
   150→      setValue(current)
   151→      if (progress < 1) {
   152→        rafRef.current = requestAnimationFrame(animate)
   153→      } else {
   154→        prevTargetRef.current = target
   155→      }
   156→    }
   157→    rafRef.current = requestAnimationFrame(animate)
   158→    return () => cancelAnimationFrame(rafRef.current)
   159→  }, [target, duration, enabled])
   160→
   161→  return value
   162→}

   164→// ===== Circular Progress Component =====
   165→
   166→function CircularProgress({
   167→  value,
   168→  size = 48,
   169→  strokeWidth = 4,
   170→}: {
   171→  value: number
   172→  size?: number
   173→  strokeWidth?: number
   174→}) {
   175→  const radius = (size - strokeWidth) / 2
   176→  const circumference = 2 * Math.PI * radius
   177→  const offset = circumference - (value / 100) * circumference
   178→
   179→  const color =
   180→    value >= 80 ? '#10b981' : value >= 60 ? '#eab308' : value >= 40 ? '#f97316' : '#ef4444'
   181→
   182→  return (
   183→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   184→      <svg width={size} height={size} className="-rotate-90">
   185→        <circle
   186→          cx={size / 2}
   187→          cy={size / 2}
   188→          r={radius}
   189→          fill="none"
   190→          stroke="rgba(255,255,255,0.08)"
   191→          strokeWidth={strokeWidth}
   192→        />
   193→        <circle
   194→          cx={size / 2}
   195→          cy={size / 2}
   196→          r={radius}
   197→          fill="none"
   198→          stroke={color}
   199→          strokeWidth={strokeWidth}
   200→          strokeDasharray={circumference}
   201→          strokeDashoffset={offset}
   202→          strokeLinecap="round"
   203→          className="transition-all duration-700 ease-out"
   204→        />
   205→      </svg>
   206→      <span className="absolute text-xs font-bold" style={{ color }}>
   207→        {value}%
   208→      </span>
   209→    </div>
   210→  )
   211→}

   600→// ===== Mini Sparkline Component =====
   601→
   602→function MiniSparkline({
   603→  data,
   604→  color = '#10b981',
   605→  height = 28,
   606→  width = 60,
   607→}: {
   608→  data: number[]
   609→  color?: string
   610→  height?: number
   611→  width?: number
   612→}) {
   613→  if (data.length < 2) return null
   614→
   615→  const max = Math.max(...data, 1)
   616→  const min = Math.min(...data, 0)
   617→  const range = max - min || 1
   618→
   619→  const points = data
   620→    .map((v, i) => {
   621→      const x = (i / (data.length - 1)) * width
   622→      const y = height - ((v - min) / range) * (height - 4) - 2
   623→      return `${x},${y}`
   624→    })
   625→    .join(' ')
   626→
   627→  return (
   628→    <svg width={width} height={height} className="opacity-60">
   629→      <polyline fill="none" stroke={color} strokeWidth={1.5} points={points} />
   630→    </svg>
   631→  )
   632→}
