   456→function timeAgo(dateStr: string): string {
   457→  const diff = Date.now() - new Date(dateStr).getTime()
   458→  if (diff < 60_000) return 'just now'
   459→  if (diff < 3_600_000) return `${Math.floor(diff / 60_000)}m ago`
   460→  if (diff < 86_400_000) return `${Math.floor(diff / 3_600_000)}h ago`
   461→  return `${Math.floor(diff / 86_400_000)}d ago`
   462→}
   463→
   464→function timeUntil(dateStr: string): string {
   465→  const diff = new Date(dateStr).getTime() - Date.now()
   466→  if (diff < 0) return 'overdue'
   467→  if (diff < 3_600_000) return `in ${Math.floor(diff / 60_000)}m`
   468→  if (diff < 86_400_000) return `in ${Math.floor(diff / 3_600_000)}h`
   469→  return `in ${Math.floor(diff / 86_400_000)}d`
   470→}

   474→    month: 'short',

   476→    hour: '2-digit',

   543→function StatusBadge({ status }: { status: HistoryItem['status'] }) {
   544→  const map = {
   545→    completed: { label: 'Completed', className: 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20', icon: CheckCircle2 },
   546→    failed: { label: 'Failed', className: 'bg-red-500/10 text-red-400 border-red-500/20', icon: XCircle },
   547→    generating: { label: 'Generating', className: 'bg-cyan-500/10 text-cyan-400 border-cyan-500/20', icon: Loader2 },
   548→  }
   549→  const config = map[status]
   550→  const Icon = config.icon
   551→  return (
   552→    <Badge variant="outline" className={cn('gap-1 border', config.className)}>
