   496→// ===== Syntax highlighting =====
   497→// Lightweight per-line tokenizers that emit <span> elements with CSS classes.
   498→// Designed to be fast and dependency-free; not a full parser.
   499→
   500→function highlightValue(s: string): ReactNode[] {
   501→  const tokens: ReactNode[] = []
   502→  const regex = /("(?:[^"\\]|\\.)*"|'[^']*'|-?\b\d+\.?\d*\b)/g
   503→  let lastIdx = 0
   504→  let m: RegExpExecArray | null
   505→  let i = 0
   506→  while ((m = regex.exec(s)) !== null) {
   507→    if (m.index > lastIdx) tokens.push(s.slice(lastIdx, m.index))
   508→    const val = m[0]
   509→    if (val.startsWith('"') || val.startsWith("'")) {
   510→      tokens.push(<span key={i++} className="siem-syntax-string">{val}</span>)
   511→    } else {
   512→      tokens.push(<span key={i++} className="siem-syntax-number">{val}</span>)
   513→    }
   514→    lastIdx = m.index + val.length
   515→  }
   516→  if (lastIdx < s.length) tokens.push(s.slice(lastIdx))
   517→  return tokens
   518→}

   520→function highlightSigma(code: string): ReactNode {
   521→  const lines = code.split('\n')
   522→  return lines.map((line, i) => {
   523→    if (line.trimStart().startsWith('#')) {
   524→      return <div key={i}><span className="siem-syntax-comment">{line || ' '}</span></div>
   525→    }
   526→    const keyMatch = line.match(/^(\s*)([A-Za-z_][\w.]*)(\|[\w]+)*(:)(.*)$/)
   527→    if (keyMatch) {
   528→      const indent = keyMatch[1]
   529→      const key = keyMatch[2]
   530→      const modifier = keyMatch[3] || ''
   531→      const colon = keyMatch[4]
   532→      const rest = keyMatch[5]
   533→      return (
   534→        <div key={i}>
   535→          {indent}
   536→          <span className="siem-syntax-keyword">{key}</span>
   537→          {modifier && <span className="siem-syntax-operator">{modifier}</span>}
   538→          <span className="siem-syntax-operator">{colon}</span>
   539→          {highlightValue(rest)}
   540→        </div>
   541→      )
   542→    }
   543→    const listMatch = line.match(/^(\s*)(-)(\s+)(.*)$/)
   544→    if (listMatch) {
   545→      return (
   546→        <div key={i}>
   547→          {listMatch[1]}<span className="siem-syntax-operator">{listMatch[2]}</span>{listMatch[3]}
   548→          {highlightValue(listMatch[4])}
   549→        </div>
   550→      )
   551→    }
   552→    return <div key={i}>{line || ' '}</div>
   553→  })
   554→}

   563→function highlightYara(code: string): ReactNode {
   564→  const lines = code.split('\n')
   565→  return lines.map((line, i) => {
   566→    const trimmed = line.trimStart()
   567→    if (trimmed.startsWith('//')) {
   568→      return <div key={i}><span className="siem-syntax-comment">{line || ' '}</span></div>
   569→    }
   570→    const tokens: ReactNode[] = []
   571→    const regex = /(\s+)|("(?:[^"\\]|\\.)*"|'[^']*')|(\/[^/\n]+\/[gimsux]*)|(\$\w+)|(\{[^}]*\})|([A-Za-z_]\w*)|(\b\d+(?:\.\d+)?\b)|([{}():,=<>!|&])/g
   572→    let m: RegExpExecArray | null
   573→    let ti = 0
   574→    while ((m = regex.exec(line)) !== null) {
   575→      const [full, ws, str, regexVal, variable, hex, word, num, op] = m
   576→      if (ws !== undefined) { tokens.push(ws); continue }
   577→      if (str !== undefined) { tokens.push(<span key={ti++} className="siem-syntax-string">{str}</span>); continue }
   578→      if (regexVal !== undefined) { tokens.push(<span key={ti++} className="siem-syntax-string">{regexVal}</span>); continue }
   579→      if (variable !== undefined) { tokens.push(<span key={ti++} className="siem-syntax-variable">{variable}</span>); continue }
   580→      if (hex !== undefined) { tokens.push(<span key={ti++} className="siem-syntax-number">{hex}</span>); continue }
   581→      if (word !== undefined) {
   582→        if (YARA_KEYWORDS.has(word)) tokens.push(<span key={ti++} className="siem-syntax-keyword">{word}</span>)
   583→        else tokens.push(word)
   584→        continue
   585→      }
   586→      if (num !== undefined) { tokens.push(<span key={ti++} className="siem-syntax-number">{num}</span>); continue }
   587→      if (op !== undefined) { tokens.push(<span key={ti++} className="siem-syntax-operator">{op}</span>); continue }
   588→      tokens.push(full)
   589→    }
   590→    return <div key={i}>{tokens.length ? tokens : (line || ' ')}</div>
   591→  })
   592→}

   595→  'where', 'project', 'summarize', 'extend', 'join', 'limit', 'order', 'by',
   596→  'desc', 'asc', 'let', 'and', 'or', 'not', 'contains', 'has', 'startswith',
   597→  'endswith', 'matches', 'regex', 'in', 'between', 'ago', 'now', 'bin', 'count',
   598→  'sum', 'avg', 'min', 'max', 'AND', 'OR', 'NOT', 'take', 'sort', 'distinct',
   599→  'union', 'make', 'list', 'set', 'dcount', 'iff',

   602→function highlightKqlLike(code: string): ReactNode {
   603→  const lines = code.split('\n')
   604→  return lines.map((line, i) => {
   605→    if (line.trimStart().startsWith('//') || line.trimStart().startsWith('#')) {
   606→      return <div key={i}><span className="siem-syntax-comment">{line || ' '}</span></div>
   607→    }
   608→    const tokens: ReactNode[] = []
   609→    const regex = /(\s+)|("(?:[^"\\]|\\.)*"|'[^']*')|(\*|:|=|\(|\)|\||<|>)/g
   610→    let m: RegExpExecArray | null
   611→    let ti = 0
   612→    let lastIdx = 0
   613→    while ((m = regex.exec(line)) !== null) {
   614→      if (m.index > lastIdx) {
   615→        const chunk = line.slice(lastIdx, m.index)
   616→        // split chunk into words and numbers
   617→        const sub = chunk.split(/(\b\d+(?:\.\d+)?\b)/)
   618→        for (const part of sub) {
   619→          if (!part) continue
   620→          if (/^\d+(\.\d+)?$/.test(part)) {
   621→            tokens.push(<span key={ti++} className="siem-syntax-number">{part}</span>)
   622→          } else {
   623→            const words = part.split(/(\s+)/)
   624→            for (const w of words) {
   625→              if (!w) continue
   626→              if (/^\s+$/.test(w)) { tokens.push(w); continue }
   627→              if (KQL_KEYWORDS.has(w)) tokens.push(<span key={ti++} className="siem-syntax-keyword">{w}</span>)
   628→              else tokens.push(w)
   629→            }
   630→          }
   631→        }
   632→      }
   633→      const [, ws, str, op] = m
   634→      if (ws !== undefined) { tokens.push(ws); lastIdx = m.index + ws.length; continue }
   635→      if (str !== undefined) { tokens.push(<span key={ti++} className="siem-syntax-string">{str}</span>); lastIdx = m.index + str.length; continue }
   636→      if (op !== undefined) { tokens.push(<span key={ti++} className="siem-syntax-operator">{op}</span>); lastIdx = m.index + op.length; continue }
   637→      lastIdx = m.index + m[0].length
   638→    }
   639→    if (lastIdx < line.length) {
   640→      const chunk = line.slice(lastIdx)
   641→      const sub = chunk.split(/(\b\d+(?:\.\d+)?\b)/)
   642→      for (const part of sub) {
   643→        if (!part) continue
   644→        if (/^\d+(\.\d+)?$/.test(part)) {
   645→          tokens.push(<span key={ti++} className="siem-syntax-number">{part}</span>)
   646→        } else {
   647→          const words = part.split(/(\s+)/)
   648→          for (const w of words) {
   649→            if (!w) continue
   650→            if (/^\s+$/.test(w)) { tokens.push(w); continue }
   651→            if (KQL_KEYWORDS.has(w)) tokens.push(<span key={ti++} className="siem-syntax-keyword">{w}</span>)
   652→            else tokens.push(w)
   653→          }
   654→        }
   655→      }
   656→    }
   657→    return <div key={i}>{tokens.length ? tokens : (line || ' ')}</div>
   658→  })
   659→}

   662→  if (lang === 'sigma') return highlightSigma(code)
   663→  if (lang === 'yara') return highlightYara(code)
   664→  return highlightKqlLike(code)

   729→// ===== Tiny deterministic sparkline (preserved from original) =====
   730→function hashString(s: string): number {
   731→  let h = 2166136261
   732→  for (let i = 0; i < s.length; i++) {
   733→    h ^= s.charCodeAt(i)
   734→    h = (h * 16777619) >>> 0
   735→  }
   736→  return h
   737→}

   739→function generateSparklinePoints(ruleId: string, hitCount: number): number[] {
   740→  if (hitCount <= 0) return [0, 0, 0, 0, 0, 0, 0, 0]
   741→  const seed = hashString(ruleId)
   742→  const points: number[] = []
   743→  let prev = hitCount * 0.4
   744→  for (let i = 0; i < 8; i++) {
   745→    const r = ((seed >> (i * 3)) & 0xff) / 255 - 0.45
   746→    prev = Math.max(0, prev + r * hitCount * 0.25)
   747→    points.push(prev)
   748→  }
   749→  points[points.length - 1] = hitCount
   750→  return points
   751→}
   752→
   753→function HitsSparkline({ ruleId, hitCount }: { ruleId: string; hitCount: number }) {
   754→  const points = useMemo(
   755→    () => generateSparklinePoints(ruleId, hitCount),
   756→    [ruleId, hitCount]
   757→  )
   758→  const max = Math.max(1, ...points)
   759→  const w = 44
   760→  const h = 14
   761→  const stepX = w / (points.length - 1)
   762→  const path = points
   763→    .map((p, i) => {
   764→      const x = i * stepX
   765→      const y = h - (p / max) * (h - 2) - 1
   766→      return `${i === 0 ? 'M' : 'L'}${x.toFixed(1)},${y.toFixed(1)}`
   767→    })
   768→    .join(' ')
   769→  const areaPath = `${path} L${w},${h} L0,${h} Z`
   770→  const color =
   771→    hitCount > 100 ? '#ef4444' : hitCount > 20 ? '#f97316' : hitCount > 0 ? '#10b981' : '#52525b'
   772→
   773→  if (hitCount === 0) {
   774→    return (
   775→      <svg width={w} height={h} className="shrink-0 opacity-50" aria-hidden="true">
   776→        <line x1="0" y1={h - 1} x2={w} y2={h - 1} stroke={color} strokeWidth="1" strokeDasharray="2 2" />
   777→      </svg>
   778→    )
   779→  }
   780→
   781→  return (
   782→    <svg width={w} height={h} className="shrink-0" aria-hidden="true">
   783→      <defs>
   784→        <linearGradient id={`spark-${ruleId}`} x1="0" y1="0" x2="0" y2="1">
   785→          <stop offset="0%" stopColor={color} stopOpacity="0.35" />
   786→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   787→        </linearGradient>
   788→      </defs>
   789→      <path d={areaPath} fill={`url(#spark-${ruleId})`} />
   790→      <path d={path} fill="none" stroke={color} strokeWidth="1.25" strokeLinecap="round" strokeLinejoin="round" />
   791→    </svg>
   792→  )
   793→}
