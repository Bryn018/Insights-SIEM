   109→   109→function timeAgo(dateStr: string): string {
   110→   110→  const now = Date.now()
   111→   111→  const then = new Date(dateStr).getTime()
   112→   112→  const diff = now - then
   113→   113→  if (diff < 60_000) return 'just now'
   114→   114→  if (diff < 3_600_000) return `${Math.floor(diff / 60_000)}m ago`
   115→   115→  if (diff < 86_400_000) return `${Math.floor(diff / 3_600_000)}h ago`
   116→   116→  return `${Math.floor(diff / 86_400_000)}d ago`
   117→   117→}

   125→   125→function useAnimatedCounter(target: number, duration = 1200, enabled = true) {
   126→   126→  const [value, setValue] = useState(0)
   127→   127→  const rafRef = useRef<number>(0)
   128→   128→  const prevTargetRef = useRef(0)
   129→   129→
   130→   130→  useEffect(() => {
   131→   131→    if (!enabled) {
   132→   132→      // Directly use target when animation is disabled
   133→   133→      prevTargetRef.current = target
   134→   134→      return
   135→   135→    }
   136→   136→
   137→   137→    const startVal = prevTargetRef.current
   138→   138→    const start = performance.now()
   139→   139→    const animate = (now: number) => {
   140→   140→      const elapsed = now - start
   141→   141→      const progress = Math.min(elapsed / duration, 1)
   142→   142→      // easeOutCubic
   143→   143→      const eased = 1 - Math.pow(1 - progress, 3)
   144→   144→      const current = Math.round(startVal + eased * (target - startVal))
   145→   145→      setValue(current)
   146→   146→      if (progress < 1) {
   147→   147→        rafRef.current = requestAnimationFrame(animate)
   148→   148→      } else {
   149→   149→        prevTargetRef.current = target
   150→   150→      }
   151→   151→    }
   152→   152→    rafRef.current = requestAnimationFrame(animate)
   153→   153→    return () => cancelAnimationFrame(rafRef.current)
   154→   154→  }, [target, duration, enabled])
   155→   155→
   156→   156→  return value
   157→   157→}

   159→   159→// ===== Circular Progress Component =====
   160→   160→
   161→   161→function CircularProgress({
   162→   162→  value,
   163→   163→  size = 48,
   164→   164→  strokeWidth = 4,
   165→   165→}: {
   166→   166→  value: number
   167→   167→  size?: number
   168→   168→  strokeWidth?: number
   169→   169→}) {
   170→   170→  const radius = (size - strokeWidth) / 2
   171→   171→  const circumference = 2 * Math.PI * radius
   172→   172→  const offset = circumference - (value / 100) * circumference
   173→   173→
   174→   174→  const color =
   175→   175→    value >= 80 ? '#10b981' : value >= 60 ? '#eab308' : value >= 40 ? '#f97316' : '#ef4444'
   176→   176→
   177→   177→  return (
   178→   178→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   179→   179→      <svg width={size} height={size} className="-rotate-90">
   180→   180→        <circle
   181→   181→          cx={size / 2}
   182→   182→          cy={size / 2}
   183→   183→          r={radius}
   184→   184→          fill="none"
   185→   185→          stroke="rgba(255,255,255,0.08)"
   186→   186→          strokeWidth={strokeWidth}
   187→   187→        />
   188→   188→        <circle
   189→   189→          cx={size / 2}
   190→   190→          cy={size / 2}
   191→   191→          r={radius}
   192→   192→          fill="none"
   193→   193→          stroke={color}
   194→   194→          strokeWidth={strokeWidth}
   195→   195→          strokeDasharray={circumference}
   196→   196→          strokeDashoffset={offset}
   197→   197→          strokeLinecap="round"
   198→   198→          className="transition-all duration-700 ease-out"
   199→   199→        />
   200→   200→      </svg>
   201→   201→      <span className="absolute text-xs font-bold" style={{ color }}>
   202→   202→        {value}%
   203→   203→      </span>
   204→   204→    </div>
   205→   205→  )
   206→   206→}

   476→   476→// ===== Mini Sparkline Component =====
   477→   477→
   478→   478→function MiniSparkline({
   479→   479→  data,
   480→   480→  color = '#10b981',
   481→   481→  height = 28,
   482→   482→  width = 60,
   483→   483→}: {
   484→   484→  data: number[]
   485→   485→  color?: string
   486→   486→  height?: number
   487→   487→  width?: number
   488→   488→}) {
   489→   489→  if (data.length < 2) return null
   490→   490→
   491→   491→  const max = Math.max(...data, 1)
   492→   492→  const min = Math.min(...data, 0)
   493→   493→  const range = max - min || 1
   494→   494→
   495→   495→  const points = data
   496→   496→    .map((v, i) => {
   497→   497→      const x = (i / (data.length - 1)) * width
   498→   498→      const y = height - ((v - min) / range) * (height - 4) - 2
   499→   499→      return `${x},${y}`
   500→   500→    })
   501→   501→    .join(' ')
   502→   502→
   503→   503→  return (
   504→   504→    <svg width={width} height={height} className="opacity-60">
   505→   505→      <polyline fill="none" stroke={color} strokeWidth={1.5} points={points} />
   506→   506→    </svg>
   507→   507→  )
   508→   508→}
