   159→function AssetStatsCards({ assets, atRiskCount }: { assets: Asset[]; atRiskCount: number }) {
   160→  const stats = useMemo(() => {
   161→    const total = assets.length
   162→    const active = assets.filter((a) => a.status === 'active').length
   163→    const critical = assets.filter((a) => a.criticality === 'critical').length
   164→    return { total, active, critical, atRisk: atRiskCount }
   165→  }, [assets, atRiskCount])
   166→
   167→  const cards = [
   168→    {
   169→      label: 'Total Assets',
   170→      value: stats.total,
   171→      icon: Network,
   172→      color: 'text-zinc-200',
   173→      bg: 'bg-zinc-800/50',
   174→      border: 'border-zinc-700',
   175→    },
   176→    {
   177→      label: 'Active',
   178→      value: stats.active,
   179→      icon: Activity,
   180→      color: 'text-emerald-400',
   181→      bg: 'bg-emerald-500/5',
   182→      border: 'border-emerald-500/20',
   183→    },
   184→    {
   185→      label: 'Critical',
   186→      value: stats.critical,
   187→      icon: AlertTriangle,
   188→      color: 'text-red-400',
   189→      bg: 'bg-red-500/5',
   190→      border: 'border-red-500/20',
   191→    },
   192→    {
   193→      label: 'At Risk',
   194→      value: stats.atRisk,
   195→      icon: ShieldAlert,
   196→      color: 'text-amber-400',
   197→      bg: 'bg-amber-500/5',
   198→      border: 'border-amber-500/20',
   199→    },
   200→  ]
   201→
   202→  return (
   203→    <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   204→      {cards.map((card) => (
   205→        <Card
   206→          key={card.label}
   207→          className={cn('rounded-xl border p-4 transition-colors hover:border-opacity-60', card.bg, card.border)}
   208→        >
   209→          <div className="flex items-center justify-between">
   210→            <span className="text-[10px] font-semibold uppercase tracking-wider text-zinc-500">
   211→              {card.label}
   212→            </span>
   213→            <card.icon className={cn('h-4 w-4', card.color)} />
   214→          </div>
   215→          <div className={cn('mt-1 text-2xl font-bold tabular-nums', card.color)}>
   216→            {card.value.toLocaleString()}
   217→          </div>
   218→        </Card>
   219→      ))}
   220→    </div>
   221→  )
   222→}

   224→// ===== Filter Toggle Button =====
   225→
   226→function FilterToggle({
   227→  label,
   228→  options,
   229→  selected,
   230→  onToggle,
   231→}: {
   232→  label: string
   233→  options: { value: string; label: string }[]
   234→  selected: string[]
   235→  onToggle: (value: string) => void
   236→}) {
   237→  return (
   238→    <div className="flex flex-wrap items-center gap-1">
   239→      <span className="text-[10px] font-semibold uppercase tracking-wider text-zinc-500 mr-1">
   240→        {label}:
   241→      </span>
   242→      {options.map((opt) => {
   243→        const isActive = selected.includes(opt.value)
   244→        return (
   245→          <Button
   246→            key={opt.value}
   247→            variant="ghost"
   248→            size="sm"
   249→            className={cn(
   250→              'h-6 px-2 text-[10px] rounded-full border transition-colors',
   251→              isActive
   252→                ? 'bg-emerald-600/20 text-emerald-400 border-emerald-500/30'
   253→                : 'text-zinc-500 border-zinc-700 hover:text-zinc-300 hover:border-zinc-600'
   254→            )}
   255→            onClick={() => onToggle(opt.value)}
   256→          >
   257→            {opt.label}
   258→          </Button>
   259→        )
   260→      })}
   261→    </div>
   262→  )
   263→}
