   196→function syntaxHighlightJson(json: Record<string, unknown>): React.ReactNode[] {
   197→  const str = JSON.stringify(json, null, 2)
   198→  const lines = str.split('\n')
   199→  return lines.map((line, i) => {
   200→    // Match key-value pairs
   201→    const keyMatch = line.match(/^(\s*)"([^"]+)":/)
   202→    if (keyMatch) {
   203→      const [, indent, key] = keyMatch
   204→      const rest = line.slice(keyMatch[0].length)
   205→      return (
   206→        <div key={i}>
   207→          <span>{indent}</span>
   208→          <span className="text-emerald-400">"{key}"</span>
   209→          <span className="text-zinc-500">:</span>
   210→          {highlightValue(rest)}
   211→        </div>
   212→      )
   213→    }
   214→    // Match array values and closing braces
   215→    return <div key={i}>{highlightValue(line)}</div>
   216→  })
   217→}
   218→
   219→function highlightValue(str: string): React.ReactNode {
   220→  // String values
   221→  const stringMatch = str.match(/^ (.*)"([^"]*)"(,?)$/)
   222→  if (stringMatch) {
   223→    const [, space, val, comma] = stringMatch
   224→    return (
   225→      <>
   226→        {space}<span className="text-amber-300">"{val}"</span>{comma && <span className="text-zinc-500">{comma}</span>}
   227→      </>
   228→    )
   229→  }
   230→  // Number values
   231→  const numberMatch = str.match(/^ (\d+)(,?)$/)
   232→  if (numberMatch) {
   233→    const [, num, comma] = numberMatch
   234→    return (
   235→      <>
   236→        <span className="text-cyan-300"> {num}</span>{comma && <span className="text-zinc-500">{comma}</span>}
   237→      </>
   238→    )
   239→  }
   240→  // Boolean values
   241→  if (str.includes('true') || str.includes('false')) {
   242→    return <span className="text-purple-300">{str}</span>
   243→  }
   244→  // null
   245→  if (str.includes('null')) {
   246→    return <span className="text-red-300">{str}</span>
   247→  }
   248→  // Brackets / braces
   249→  return <span className="text-zinc-400">{str}</span>
   250→}

   261→// ===== Log Field Syntax Highlighting =====
   262→// Wraps common log artifacts (IPs, URLs, severity keywords, numbers) in
   263→// colored spans. Operates on plain strings; safe to chain after search-term
   264→// highlighting by applying it to the raw message and passing the search
   265→// query through unchanged.
   266→
   267→const IP_REGEX = /\b(?:\d{1,3}\.){3}\d{1,3}\b/g
   268→const URL_REGEX = /\bhttps?:\/\/[^\s<>"]+/gi
   269→const SEVERITY_KEYWORDS = /\b(ERROR|FAILED|FAILURE|FAIL|CRITICAL|DROP|REJECT|OOM|DENIED|BLOCKED|WARN|WARNING|SLOW|ALERT|INFO|DEBUG|NOTICE)\b/g
   270→const NUM_REGEX = /\b(?<![\w.])(\d{2,6})(?![\w.])\b/g
   271→
   272→function highlightLogFields(
   273→  text: string,
   274→  query: string,
   275→  options: { highlightSearch?: boolean } = {}
   276→): React.ReactNode {
   277→  const { highlightSearch = true } = options
   278→
   279→  // First, if there's a search query, isolate the matched regions so we
   280→  // don't double-colorize inside <mark> tags.
   281→  if (highlightSearch && query.trim()) {
   282→    try {
   283→      const searchRegex = new RegExp(`(${query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi')
   284→      const parts = text.split(searchRegex)
   285→      return (
   286→        <>
   287→          {parts.map((part, i) =>
   288→            searchRegex.test(part) ? (
   289→              <mark key={i} className="bg-yellow-400/25 text-yellow-200 rounded px-0.5">{part}</mark>
   290→            ) : (
   291→              <span key={i}>{colorizeLogText(part)}</span>
   292→            )
   293→          )}
   294→        </>
   295→      )
   296→    } catch {
   297→      // fall through to plain colorize
   298→    }
   299→  }
   300→
   301→  return colorizeLogText(text)
   302→}
   303→
   304→function colorizeLogText(text: string): React.ReactNode[] {
   305→  // Tokenize by collecting all regex matches with their positions, then
   306→  // walking through the string and emitting colored spans for matches and
   307→  // plain spans for the gaps.
   308→  type Match = { start: number; end: number; text: string; cls: string }
   309→  const matches: Match[] = []
   310→
   311→  const collect = (re: RegExp, cls: string) => {
   312→    re.lastIndex = 0
   313→    let m: RegExpExecArray | null
   314→    while ((m = re.exec(text)) !== null) {
   315→      matches.push({ start: m.index, end: m.index + m[0].length, text: m[0], cls })
   316→      // Prevent infinite loop on zero-length matches
   317→      if (m[0].length === 0) re.lastIndex++
   318→    }
   319→  }
   320→
   321→  // Order matters: URLs first (so we don't double-color IPs inside URLs),
   322→  // then IPs, then severity keywords, then numbers.
   323→  collect(URL_REGEX, 'siem-log-url')
   324→  collect(IP_REGEX, 'siem-log-ip')
   325→  collect(SEVERITY_KEYWORDS, 'siem-log-sev')
   326→  collect(NUM_REGEX, 'siem-log-num')
   327→
   328→  // Remove overlaps (earlier-collected = higher priority)
   329→  matches.sort((a, b) => a.start - b.start || a.end - b.end)
   330→  const filtered: Match[] = []
   331→  let lastEnd = 0
   332→  for (const m of matches) {
   333→    if (m.start >= lastEnd) {
   334→      filtered.push(m)
   335→      lastEnd = m.end
   336→    }
   337→  }
   338→
   339→  const nodes: React.ReactNode[] = []
   340→  let cursor = 0
   341→  filtered.forEach((m, i) => {
   342→    if (m.start > cursor) {
   343→      nodes.push(<span key={`txt-${i}`}>{text.slice(cursor, m.start)}</span>)
   344→    }
   345→    nodes.push(<span key={`hl-${i}`} className={m.cls}>{m.text}</span>)
   346→    cursor = m.end
   347→  })
   348→  if (cursor < text.length) {
   349→    nodes.push(<span key="txt-tail">{text.slice(cursor)}</span>)
   350→  }
   351→  return nodes
   352→}

   356→function LogRow({
   357→  log,
   358→  isExpanded,
   359→  onToggle,
   360→  query,
   361→  lineNum,
   362→}: {
   363→  log: WsLogEvent
   364→  isExpanded: boolean

   366→  query: string

   370→  const LevelIcon = levelCfg.icon
   371→  const ts = new Date(log.timestamp)
   372→  const relativeTime = formatDistanceToNow(ts, { addSuffix: true })
   373→  const absoluteTime = format(ts, 'yyyy-MM-dd HH:mm:ss.SSS')
   374→
   375→  return (

   403→        {/* Timestamp */}

   413→                <div>Relative: {relativeTime}</div>

   417→          </Tooltip>
   418→        </TooltipProvider>
   419→
   420→        {/* Level */}
   421→        <span className={cn('shrink-0 w-16 font-semibold flex items-center gap-1', levelCfg.text)}>
   422→          <LevelIcon className="h-3 w-3" />
   423→          {(log.level || 'info').toUpperCase()}
   424→        </span>
   425→
   426→        {/* Hostname */}
   427→        <span className="shrink-0 w-28 truncate text-zinc-400" title={log.hostname}>
   428→          {log.hostname || '—'}
   429→        </span>
   430→
   431→        {/* Source/Service */}
   432→        <span className="shrink-0 w-20 truncate text-emerald-400/70" title={log.source || log.service}>
   433→          {log.source || log.service}
   434→        </span>
   435→
   436→        {/* Message (with field highlighting + search highlight) */}
   437→        <span className="flex-1 truncate text-zinc-300 min-w-0">
   438→          {highlightLogFields(
   439→            log.message.length > 200 ? log.message.slice(0, 200) + '...' : log.message,
   440→            query
   441→          )}
   442→        </span>
   443→      </div>
   444→
   445→      {/* Expanded Detail */}
   446→      <AnimatePresence>
   447→        {isExpanded && (
   448→          <motion.div
   449→            initial={{ opacity: 0, height: 0 }}
   450→            animate={{ opacity: 1, height: 'auto' }}
   451→            exit={{ opacity: 0, height: 0 }}
   452→            transition={{ duration: 0.15 }}
   453→            className="border-b border-zinc-800 bg-zinc-950/80 overflow-hidden"
   454→          >
   455→            <div className="p-4 space-y-3">
   456→              {/* Full Message */}
   457→              <div>
   458→                <span className="text-[10px] uppercase font-semibold text-zinc-500 tracking-wider">Full Message</span>
   459→                <div className="mt-1 text-xs font-mono text-zinc-300 leading-relaxed whitespace-pre-wrap break-all">
   460→                  {highlightLogFields(log.message, query)}
   461→                </div>
   462→              </div>
   463→
   464→              {/* JSON Document */}
   465→              {log.raw && (
   466→                <div>
   467→                  <span className="text-[10px] uppercase font-semibold text-zinc-500 tracking-wider">Document JSON</span>
   468→                  <div className="mt-1 max-h-64 overflow-auto rounded-md bg-zinc-900 border border-zinc-800 p-3 text-[11px] font-mono leading-relaxed custom-scrollbar">
   469→                    {syntaxHighlightJson(log.raw)}
   470→                  </div>
   471→                </div>
   472→              )}
   473→
   474→              {/* Quick Fields */}
   475→              <div className="flex flex-wrap gap-3 text-[10px]">
   476→                <span className="text-zinc-500">Timestamp: <span className="text-zinc-300">{absoluteTime}</span></span>
   477→                <span className="text-zinc-500">Host: <span className="text-zinc-300">{log.hostname}</span></span>
   478→                <span className="text-zinc-500">Service: <span className="text-zinc-300">{log.service}</span></span>
   479→                <span className="text-zinc-500">Level: <span className={levelCfg.text}>{log.level}</span></span>
   480→              </div>
   481→            </div>
   482→          </motion.div>
   483→        )}
   484→      </AnimatePresence>
   485→    </>
   486→  )
   487→}

   495→  onToggle,
   496→}: {
   497→  logs: WsLogEvent[]
   498→  onFieldClick: (field: string, value: string) => void
   499→  visible: boolean
   500→  onToggle: () => void
   501→}) {
   502→  const fields = useMemo(() => {
   503→    const result: Record<string, Record<string, number>> = {}
   504→
   505→    // Analyze top-level fields
   506→    const analyzers: Record<string, (log: WsLogEvent) => string | null> = {
   507→      'host.name': (log) => log.hostname || null,
   508→      'service': (log) => log.service || null,
   509→      'log.level': (log) => log.level || null,
   510→      'source': (log) => log.source || null,
   511→    }
   512→
   513→    // Also analyze raw JSON fields
   514→    if (logs.length > 0 && logs[0].raw) {
   515→      const sampleRaw = logs[0].raw as Record<string, unknown>
   516→      const nestedPaths: string[] = []
   517→
   518→      const findPaths = (obj: Record<string, unknown>, prefix = '') => {
   519→        for (const [k, v] of Object.entries(obj)) {
   520→          const path = prefix ? `${prefix}.${k}` : k
   521→          if (typeof v === 'string' || typeof v === 'number' || typeof v === 'boolean') {
   522→            if (!nestedPaths.includes(path)) nestedPaths.push(path)
   523→          } else if (typeof v === 'object' && v !== null && !Array.isArray(v)) {
   524→            findPaths(v as Record<string, unknown>, path)
   525→          }
   526→        }
   527→      }
   528→
   529→      findPaths(sampleRaw)
   530→      nestedPaths.slice(0, 10).forEach((path) => {
   531→        analyzers[path] = (log) => {
   532→          if (!log.raw) return null
   533→          let current: unknown = log.raw
   534→          for (const key of path.split('.')) {
   535→            if (current && typeof current === 'object') {
   536→              current = (current as Record<string, unknown>)[key]
   537→            } else {
   538→              return null
   539→            }
   540→          }
   541→          return current != null ? String(current) : null
   542→        }
   543→      })
   544→    }
   545→
   546→    for (const [fieldName, analyzer] of Object.entries(analyzers)) {
   547→      const counts: Record<string, number> = {}
   548→      for (const log of logs) {
   549→        const val = analyzer(log)
   550→        if (val) counts[val] = (counts[val] || 0) + 1
   551→      }
   552→      if (Object.keys(counts).length > 0) {
   553→        result[fieldName] = counts
   554→      }
   555→    }
   556→
   557→    return result
   558→  }, [logs])
   559→
   560→  if (!visible) return null
   561→
   562→  return (
   563→    <motion.div
   564→      initial={{ width: 0, opacity: 0 }}
   565→      animate={{ width: 280, opacity: 1 }}
   566→      exit={{ width: 0, opacity: 0 }}
   567→      transition={{ duration: 0.2 }}
   568→      className="shrink-0 border-l border-zinc-800 bg-zinc-900/50 overflow-y-auto custom-scrollbar"
   569→    >
   570→      <div className="p-3 space-y-4">
   571→        <div className="flex items-center justify-between">
   572→          <h3 className="text-xs font-semibold text-zinc-300 uppercase tracking-wider">Field Statistics</h3>
   573→          <Button variant="ghost" size="icon" className="h-5 w-5" onClick={onToggle}>
   574→            <PanelRightClose className="h-3 w-3" />
   575→          </Button>
   576→        </div>
   577→
   578→        {Object.entries(fields).map(([fieldName, counts]) => {
   579→          const sorted = Object.entries(counts).sort((a, b) => b[1] - a[1])
   580→          const maxCount = sorted[0]?.[1] || 1
   581→
   582→          return (
   583→            <div key={fieldName}>
   584→              <span className="text-[10px] font-semibold uppercase text-emerald-400/80 tracking-wider">{fieldName}</span>
   585→              <div className="mt-1.5 space-y-1">
   586→                {sorted.slice(0, 6).map(([val, count]) => (
   587→                  <button
   588→                    key={val}
   589→                    className="w-full text-left group"
   590→                    onClick={() => onFieldClick(fieldName, val)}
   591→                  >
   592→                    <div className="flex items-center justify-between text-[10px]">
   593→                      <span className="truncate text-zinc-300 group-hover:text-emerald-300 transition-colors max-w-[140px]">{val}</span>
   594→                      <span className="text-zinc-500 tabular-nums">{count}</span>
   595→                    </div>
   596→                    <div className="mt-0.5 h-1 rounded-full bg-zinc-800 overflow-hidden">
   597→                      <div
   598→                        className="h-full rounded-full bg-emerald-500/30 group-hover:bg-emerald-500/50 transition-colors"
   599→                        style={{ width: `${(count / maxCount) * 100}%` }}
   600→                      />
   601→                    </div>
   602→                  </button>
   603→                ))}
   604→              </div>
   605→            </div>
   606→          )
   607→        })}
   608→      </div>
   609→    </motion.div>
   610→  )
   611→}
