   235→// ===== Helper: Parse JSON setting value =====
   236→
   237→function parseSettingValue(value: string | null | undefined, fallback: string = ''): string {
   238→  if (!value) return fallback
   239→  try {
   240→    const parsed = JSON.parse(value)
   241→    if (typeof parsed === 'object' && parsed !== null && 'value' in parsed) {
   242→      return String((parsed as { value: unknown }).value)
   243→    }
   244→    return String(parsed)
   245→  } catch {
   246→    return value
   247→  }
   248→}
