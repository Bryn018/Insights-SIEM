   206→// ===== Helpers =====

   208→function formatBytes(bytes: number): string {
   209→  if (bytes < 1024) return `${bytes} B`
   210→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`
   211→  if (bytes < 1073741824) return `${(bytes / 1048576).toFixed(1)} MB`
   212→  return `${(bytes / 1073741824).toFixed(1)} GB`
   213→}
   214→
   215→function formatBytesShort(bytes: number): string {
   216→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(0)}K`
   217→  if (bytes < 1073741824) return `${(bytes / 1048576).toFixed(1)}M`
   218→  return `${(bytes / 1073741824).toFixed(1)}G`
   219→}

   286→function BandwidthSparkline() {
   287→  const w = 600
   288→  const h = 80
   289→  const padX = 4
   290→  const padY = 8
   291→  const max = Math.max(...BANDWIDTH_DATA)
   292→  const min = Math.min(...BANDWIDTH_DATA)
   293→  const range = max - min || 1
   294→
   295→  const points = BANDWIDTH_DATA.map((v, i) => {
   296→    const x = padX + (i / (BANDWIDTH_DATA.length - 1)) * (w - padX * 2)
   297→    const y = padY + (1 - (v - min) / range) * (h - padY * 2)
   298→    return `${x},${y}`
   299→  }).join(' ')
   300→
   301→  const areaPoints = `${padX},${h - padY} ${points} ${w - padX},${h - padY}`
   302→
   303→  return (
   304→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full h-20" preserveAspectRatio="none">
   305→      <defs>
   306→        <linearGradient id="bwGrad" x1="0" y1="0" x2="0" y2="1">
   307→          <stop offset="0%" stopColor="#10b981" stopOpacity="0.3" />
   308→          <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
   309→        </linearGradient>
   310→      </defs>
   311→      <polygon points={areaPoints} fill="url(#bwGrad)" />
   312→      <polyline points={points} fill="none" stroke="#10b981" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
   313→      {/* Highlight last point */}
   314→      {(() => {
   315→        const lastIdx = BANDWIDTH_DATA.length - 1
   316→        const lx = padX + (lastIdx / (BANDWIDTH_DATA.length - 1)) * (w - padX * 2)
   317→        const ly = padY + (1 - (BANDWIDTH_DATA[lastIdx] - min) / range) * (h - padY * 2)
   318→        return <circle cx={lx} cy={ly} r="3" fill="#10b981" className="animate-pulse" />
   319→      })()}
   320→    </svg>
   321→  )
   322→}

   411→
   412→  const filteredFlows = useMemo(() => {
   413→    let flows = [...MOCK_FLOWS]
   414→
   415→    // Protocol filter
   416→    if (filterProtocol !== 'all') {
   417→      if (filterProtocol === 'HTTP/HTTPS') {
   418→        flows = flows.filter(f => f.protocol === 'HTTP' || f.protocol === 'HTTPS')
   419→      } else {
   420→        flows = flows.filter(f => f.protocol === filterProtocol)
   421→      }
   422→    }
   423→
   424→    // Search filter
   425→    if (searchQuery.trim()) {
   426→      const q = searchQuery.toLowerCase().trim()
   427→      flows = flows.filter(f =>
   428→        f.sourceIp.includes(q) ||
   429→        f.destIp.includes(q) ||
   430→        String(f.sourcePort).includes(q) ||
   431→        String(f.destPort).includes(q) ||
   432→        f.protocol.toLowerCase().includes(q)
   433→      )
   434→    }
   435→
   436→    // Sort
   437→    flows.sort((a, b) => {
   438→      let cmp = 0
   439→      switch (sortCol) {
   440→        case 'sourceIp': cmp = a.sourceIp.localeCompare(b.sourceIp); break
   441→        case 'destIp': cmp = a.destIp.localeCompare(b.destIp); break
   442→        case 'protocol': cmp = a.protocol.localeCompare(b.protocol); break
   443→        case 'bytes': cmp = a.bytes - b.bytes; break
   444→        case 'packets': cmp = a.packets - b.packets; break
   445→        case 'risk': {
   446→          const riskOrder: Record<Risk, number> = { Critical: 0, High: 1, Medium: 2, Low: 3 }
   447→          cmp = riskOrder[a.risk] - riskOrder[b.risk]
   448→          break
   449→        }
   450→        default: cmp = 0
   451→      }
   452→      return sortDir === 'asc' ? cmp : -cmp
   453→    })
   454→
   455→    return flows
   456→  }, [filterProtocol, searchQuery, sortCol, sortDir])

   502→        <StatCard icon={Wifi} iconColor="text-cyan-400" iconBg="bg-cyan-500/10 border border-cyan-500/20" value="847 Mbps" label="Bandwidth" trend="+12.5%" trendUp={true} />
