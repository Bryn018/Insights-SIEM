    93→function generateTimeSeries(length: number, base: number, variance: number): Array<{ t: string; v: number }> {
    94→  const now = Date.now()
    95→  return Array.from({ length }, (_, i) => {
    96→    const ts = new Date(now - (length - 1 - i) * 5000)
    97→    return {
    98→      t: ts.toLocaleTimeString('en-US', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' }),
    99→      v: Math.max(0, Math.round(base + (Math.random() - 0.5) * variance * 2)),
   100→    }
   101→  })
   102→}
   103→
   104→function generateLatencySeries(length: number): Array<{ t: string; p50: number; p95: number; p99: number }> {
   105→  const now = Date.now()
   106→  return Array.from({ length }, (_, i) => {
   107→    const ts = new Date(now - (length - 1 - i) * 5000)
   108→    const p50 = 12 + Math.round(Math.random() * 8)
   109→    return {
   110→      t: ts.toLocaleTimeString('en-US', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' }),
   111→      p50,
   112→      p95: p50 + 15 + Math.round(Math.random() * 10),
   113→      p99: p50 + 45 + Math.round(Math.random() * 20),
   114→    }
   115→  })
   116→}

   186→// ===== Circular Gauge SVG =====
   187→
   188→function CircularGauge({ value, size = 80, strokeWidth = 6, label, unit, color }: {
   189→  value: number
   190→  size?: number
   191→  strokeWidth?: number
   192→  label: string
   193→  unit: string
   194→  color?: string
   195→}) {
   196→  const radius = (size - strokeWidth) / 2
   197→  const circumference = 2 * Math.PI * radius
   198→  const offset = circumference - (Math.min(value, 100) / 100) * circumference
   199→
   200→  const gaugeColor = color || (
   201→    value >= 80 ? '#ef4444' : value >= 60 ? '#f97316' : value >= 40 ? '#eab308' : '#10b981'
   202→  )
   203→
   204→  return (
   205→    <div className="flex flex-col items-center gap-1">
   206→      <div className="relative" style={{ width: size, height: size }}>
   207→        <svg width={size} height={size} className="-rotate-90">
   208→          <circle
   209→            cx={size / 2}
   210→            cy={size / 2}
   211→            r={radius}
   212→            fill="none"
   213→            stroke="rgba(255,255,255,0.06)"
   214→            strokeWidth={strokeWidth}
   215→          />
   216→          <circle
   217→            cx={size / 2}
   218→            cy={size / 2}
   219→            r={radius}
   220→            fill="none"
   221→            stroke={gaugeColor}
   222→            strokeWidth={strokeWidth}
   223→            strokeDasharray={circumference}
   224→            strokeDashoffset={offset}
   225→            strokeLinecap="round"
   226→            className="transition-all duration-700 ease-out"
   227→            style={{ filter: `drop-shadow(0 0 4px ${gaugeColor}40)` }}
   228→          />
   229→        </svg>
   230→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   231→          <span className="text-base font-bold" style={{ color: gaugeColor }}>{value}</span>
   232→          <span className="text-[8px] text-zinc-500">{unit}</span>
   233→        </div>
   234→      </div>
   235→      <span className="text-[10px] text-zinc-500">{label}</span>
   236→    </div>
   237→  )
   238→}

   335→// ===== Memory Bar =====
   336→
   337→function MemoryBar({ used, cached, free, total }: { used: number; cached: number; free: number; total: number }) {
   338→  // Clamp values to prevent negative percentages or overflows
   339→  const clampedFree = Math.max(0, free)
   340→  const clampedUsed = Math.min(used, total)
   341→  const clampedCached = Math.min(cached, total - clampedUsed)
   342→  const usedPct = (clampedUsed / total) * 100
   343→  const cachedPct = (clampedCached / total) * 100
   344→  const freePct = (clampedFree / total) * 100
   345→
   346→  return (
   347→    <div className="space-y-1.5">
   348→      <div className="flex items-center justify-between text-[10px]">
   349→        <span className="text-zinc-500">Memory</span>
   350→        <span className="text-zinc-400">{clampedUsed}GB / {total}GB</span>
   351→      </div>
   352→      <div className="flex h-3 overflow-hidden rounded-full bg-zinc-800">
   353→        <motion.div
   354→          initial={{ width: 0 }}
   355→          animate={{ width: `${usedPct}%` }}
   356→          transition={{ duration: 0.8, ease: 'easeOut' }}
   357→          className="bg-emerald-500"
   358→          title={`Used: ${clampedUsed}GB`}
   359→        />
   360→        <motion.div
   361→          initial={{ width: 0 }}
   362→          animate={{ width: `${cachedPct}%` }}
   363→          transition={{ duration: 0.8, ease: 'easeOut', delay: 0.1 }}
   364→          className="bg-amber-500/60"
   365→          title={`Cached: ${clampedCached}GB`}
   366→        />
   367→        <motion.div
   368→          initial={{ width: 0 }}
   369→          animate={{ width: `${freePct}%` }}
   370→          transition={{ duration: 0.8, ease: 'easeOut', delay: 0.2 }}
   371→          className="bg-zinc-700"
   372→          title={`Free: ${clampedFree}GB`}
   373→        />
   374→      </div>
   375→      <div className="flex items-center gap-3 text-[8px] text-zinc-600">
   376→        <span className="flex items-center gap-1">
   377→          <span className="h-1.5 w-1.5 rounded-full bg-emerald-500" /> Used ({Math.round(usedPct)}%)
   378→        </span>
   379→        <span className="flex items-center gap-1">
   380→          <span className="h-1.5 w-1.5 rounded-full bg-amber-500/60" /> Cached ({Math.round(cachedPct)}%)
   381→        </span>
   382→        <span className="flex items-center gap-1">
   383→          <span className="h-1.5 w-1.5 rounded-full bg-zinc-700" /> Free ({Math.round(freePct)}%)
   384→        </span>
   385→      </div>
   386→    </div>
   387→  )
   388→}

   392→function StatusDot({ status }: { status: ServiceStatus }) {
   393→  const dotColor = status === 'healthy' ? 'bg-emerald-400' : status === 'degraded' ? 'bg-amber-400' : 'bg-red-400'

   396→  return (

   398→  )

   403→export function SystemDiagnostics() {
   404→  const [data, setData] = useState<DiagnosticsData | null>(null)
   405→  const [isOpen, setIsOpen] = useState(true)
   406→  const [isRefreshing, setIsRefreshing] = useState(false)
   407→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
   408→
   409→  const fetchDiagnostics = useCallback(async () => {
   410→    setIsRefreshing(true)
   411→    try {
   412→      // Attempt real API integration first
   413→      const res = await fetch('/api/dashboard')
   414→      if (res.ok) {
   415→        // API exists — augment with mock diagnostics data
   416→        // In production, a dedicated /api/diagnostics endpoint would provide this
   417→        const apiData = await res.json()
   418→        // We still use mock for the detailed diagnostics, but overlay real data where available
   419→        const mock = generateMockData()
   420→        // If API has integration health, map it to services
   421→        if (apiData.systemHealth?.integrationHealth) {
   422→          const ih: Record<string, number> = apiData.systemHealth.integrationHealth
   423→          mock.services = mock.services.map(s => {
   424→            const status: ServiceStatus = (ih.success ?? 0) > 0 ? 'healthy' : (ih.failure ?? 0) > 0 ? 'down' : s.status
   425→            return { ...s, status: s.name === 'Fluent Bit' ? s.status : status }
   426→          })
   427→        }
   428→        setData(mock)
   429→      } else {
   430→        setData(generateMockData())
   431→      }
   432→    } catch {
   433→      setData(generateMockData())
   434→    } finally {
   435→      setIsRefreshing(false)
   436→    }
   437→  }, [])
   438→
   439→  useEffect(() => {
   440→    fetchDiagnostics()
   441→    // Auto-refresh every 30 seconds
   442→    intervalRef.current = setInterval(fetchDiagnostics, 30000)
   443→    return () => {
   444→      if (intervalRef.current) clearInterval(intervalRef.current)
   445→    }
   446→  }, [fetchDiagnostics])
   447→
   448→  // Compute service status counts
   449→  const statusCounts = useMemo(() => {
   450→    if (!data) return { healthy: 0, degraded: 0, down: 0 }
   451→    return data.services.reduce(
   452→      (acc, s) => { acc[s.status]++; return acc },
   453→      { healthy: 0, degraded: 0, down: 0 } as Record<ServiceStatus, number>
   454→    )
   455→  }, [data])
   456→
   457→  const handleManualRefresh = useCallback(() => {
   458→    if (intervalRef.current) clearInterval(intervalRef.current)
   459→    fetchDiagnostics()
   460→    intervalRef.current = setInterval(fetchDiagnostics, 30000)
   461→  }, [fetchDiagnostics])
   462→
   463→  if (!data) {
   464→    return (
   465→      <Card className="border-zinc-800 bg-zinc-950">
   466→        <CardContent className="p-4">
   467→          <div className="flex items-center justify-center py-12 text-sm text-zinc-500">
   468→            <RefreshCw className="mr-2 h-4 w-4 animate-spin" />
   469→            Loading diagnostics...
   470→          </div>
   471→        </CardContent>
   472→      </Card>
   473→    )
   474→  }
   475→
   476→  return (
   477→    <Collapsible open={isOpen} onOpenChange={setIsOpen}>
   478→      <Card className="border-zinc-800 bg-zinc-950 overflow-hidden">
   479→        <CollapsibleTrigger asChild>
   480→          <CardHeader className="cursor-pointer select-none border-b border-zinc-800 bg-zinc-950 px-4 py-3 hover:bg-zinc-900/50 transition-colors">
   481→            <div className="flex items-center justify-between">
   482→              <div className="flex items-center gap-3">
   483→                <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-emerald-500/10">
   484→                  <Gauge className="h-4 w-4 text-emerald-400" />
   485→                </div>
   486→                <div>
   487→                  <CardTitle className="text-sm font-medium text-zinc-200">System Diagnostics</CardTitle>
   488→                  <div className="flex items-center gap-2 text-[10px] text-zinc-500">
   489→                    <span className="flex items-center gap-1">
   490→                      <CheckCircle2 className="h-2.5 w-2.5 text-emerald-400" />
   491→                      {statusCounts.healthy} healthy
   492→                    </span>
   493→                    {statusCounts.degraded > 0 && (
   494→                      <span className="flex items-center gap-1">
   495→                        <AlertTriangle className="h-2.5 w-2.5 text-amber-400" />
   496→                        {statusCounts.degraded} degraded
   497→                      </span>
   498→                    )}
   499→                    {statusCounts.down > 0 && (
   500→                      <span className="flex items-center gap-1">
   501→                        <XCircle className="h-2.5 w-2.5 text-red-400" />
   502→                        {statusCounts.down} down
   503→                      </span>
   504→                    )}
   505→                  </div>
   506→                </div>
   507→              </div>
   508→              <div className="flex items-center gap-2">
   509→                <Button
   510→                  variant="ghost"
   511→                  size="sm"
   512→                  className="h-6 w-6 p-0 text-zinc-500 hover:text-zinc-300"
