   253→// ===== Helpers =====
   254→
   255→function successRate(pb: SoarPlaybook): number {
   256→  if (pb.runCount === 0) return 0
   257→  return Math.round((pb.successCount / pb.runCount) * 100)
   258→}
   259→
   260→function formatDuration(sec: number): string {
   261→  if (sec < 60) return `${sec}s`
   262→  const m = Math.floor(sec / 60)
   263→  const s = sec % 60
   264→  if (m < 60) return s === 0 ? `${m}m` : `${m}m ${s}s`
   265→  const h = Math.floor(m / 60)
   266→  const mm = m % 60
   267→  return mm === 0 ? `${h}h` : `${h}h ${mm}m`
   268→}
   269→
   270→function formatRelative(iso: string | null): string {
   271→  if (!iso) return 'Never'
   272→  try {
   273→    return formatDistanceToNow(new Date(iso), { addSuffix: true })
   274→  } catch {
   275→    return '—'
   276→  }
   277→}

   312→// ===== Mini Step Flow (used on playbook card) =====
   313→
   314→function MiniStepFlow({ steps, isActive = false }: { steps: SoarStep[]; isActive?: boolean }) {
   315→  const max = 8
   316→  const visible = steps.slice(0, max)
   317→  const overflow = steps.length - visible.length
   318→
   319→  return (
   320→    <div className="flex items-center gap-0.5 overflow-hidden">
   321→      {visible.map((step, i) => {
   322→        const cfg = STEP_TYPE_CONFIG[step.type]
   323→        const Icon = cfg.icon
   324→        return (
   325→          <div key={step.id} className="flex items-center gap-0.5 shrink-0">
   326→            <div
   327→              className={cn(
   328→                'flex size-6 items-center justify-center rounded-md border transition-transform',
   329→                cfg.bg, cfg.border,
   330→                isActive && 'hover:scale-110'
   331→              )}
   332→              title={`${i + 1}. ${step.name} (${cfg.label})`}
   333→            >
   334→              <Icon className={cn('size-3', cfg.color)} />
   335→            </div>
   336→            {i < visible.length - 1 && (
   337→              <span className="siem-step-connector" aria-hidden="true" />
   338→            )}
   339→          </div>
   340→        )
   341→      })}
   342→      {overflow > 0 && (
   343→        <span className="ml-1 text-[10px] font-medium text-zinc-500">+{overflow}</span>
   344→      )}
   345→    </div>
   346→  )
   347→}

   713→function CreatePlaybookDialog({
   714→  open,
   715→  onOpenChange,
   716→  onCreate,
   717→}: {
   718→  open: boolean
   719→  onOpenChange: (v: boolean) => void
   720→  onCreate: (pb: SoarPlaybook) => void
   721→}) {
   722→  const [name, setName] = useState('')
   723→  const [description, setDescription] = useState('')
   724→  const [trigger, setTrigger] = useState<SoarTriggerType>('manual')
   725→  const [stepRows, setStepRows] = useState<Array<{ name: string; type: SoarStepType; action: string }>>([
   726→    { name: '', type: 'trigger', action: '' },
   727→  ])
   728→
   729→  const reset = () => {
   730→    setName('')
   731→    setDescription('')
   732→    setTrigger('manual')
   733→    setStepRows([{ name: '', type: 'trigger', action: '' }])
   734→  }
   735→
   736→  const handleClose = (v: boolean) => {
   737→    if (!v) reset()
   738→    onOpenChange(v)
   739→  }
   740→
   741→  const handleAddStep = () => {
   742→    setStepRows((r) => [...r, { name: '', type: 'action', action: '' }])
   743→  }
   744→
   745→  const handleRemoveStep = (idx: number) => {
   746→    setStepRows((r) => r.filter((_, i) => i !== idx))
   747→  }
   748→
   749→  const handleStepChange = (idx: number, field: 'name' | 'type' | 'action', value: string) => {
   750→    setStepRows((r) => r.map((row, i) => (i === idx ? { ...row, [field]: value } : row)))
   751→  }
   752→
   753→  const handleCreate = () => {
   754→    if (!name.trim()) {
   755→      toast.error('Playbook name is required')
   756→      return
   757→    }
   758→    const validSteps = stepRows.filter((s) => s.name.trim() || s.action.trim())
   759→    if (validSteps.length === 0) {
   760→      toast.error('At least one step is required')
   761→      return
   762→    }
   763→    const pb: SoarPlaybook = {
   764→      id: `pb-${Date.now().toString(36)}`,
   765→      name: name.trim(),
   766→      description: description.trim() || 'No description provided.',
   767→      trigger,
   768→      status: 'draft',
   769→      lastRun: null,
   770→      runCount: 0,
   771→      successCount: 0,
   772→      failureCount: 0,
   773→      avgDuration: 0,
   774→      steps: validSteps.map((s, i) => ({
   775→        id: `s-${i + 1}-${Date.now().toString(36)}`,
   776→        name: s.name.trim() || `Step ${i + 1}`,
   777→        type: s.type,
   778→        action: s.action.trim() || 'No action specified',
   779→        config: {},
   780→      })),
   781→    }
   782→    onCreate(pb)
   783→    handleClose(false)
   784→  }
   785→
   786→  return (
   787→    <Dialog open={open} onOpenChange={handleClose}>
   788→      <DialogContent className="max-w-2xl max-h-[88vh] overflow-y-auto bg-zinc-950 border-zinc-800">
   789→        <DialogHeader>
   790→          <DialogTitle className="flex items-center gap-2 text-zinc-100">
   791→            <Plus className="size-5 text-emerald-400" />
   792→            Create New Playbook
   793→          </DialogTitle>
   794→          <DialogDescription className="text-zinc-400">
   795→            Define a new automation playbook. New playbooks are created in <span className="text-amber-400 font-medium">Draft</span> status.
   796→          </DialogDescription>
   797→        </DialogHeader>
   798→
   799→        <div className="space-y-4 py-2">
   800→          <div className="space-y-1.5">
   801→            <Label htmlFor="pb-name" className="text-xs text-zinc-300">Name</Label>
   802→            <Input
   803→              id="pb-name"
   804→              value={name}
   805→              onChange={(e) => setName(e.target.value)}
   806→              placeholder="e.g., Insider Threat Response"
   807→              className="bg-zinc-900 border-zinc-700 text-zinc-100"
   808→            />
   809→          </div>
   810→
   811→          <div className="space-y-1.5">
   812→            <Label htmlFor="pb-desc" className="text-xs text-zinc-300">Description</Label>
   813→            <Textarea
   814→              id="pb-desc"
   815→              value={description}
   816→              onChange={(e) => setDescription(e.target.value)}
   817→              placeholder="Brief description of what this playbook does..."
   818→              rows={2}
   819→              className="bg-zinc-900 border-zinc-700 text-zinc-100 resize-none"
   820→            />
   821→          </div>
   822→
   823→          <div className="space-y-1.5">
   824→            <Label className="text-xs text-zinc-300">Trigger Type</Label>
   825→            <Select value={trigger} onValueChange={(v) => setTrigger(v as SoarTriggerType)}>
   826→              <SelectTrigger className="bg-zinc-900 border-zinc-700 text-zinc-100">
   827→                <SelectValue />
   828→              </SelectTrigger>
   829→              <SelectContent className="bg-zinc-900 border-zinc-700">
   830→                <SelectItem value="manual">Manual</SelectItem>
   831→                <SelectItem value="scheduled">Scheduled</SelectItem>
   832→                <SelectItem value="alert-based">Alert-Based</SelectItem>

  1072→    const avgDur = playbooksWithRuns.length > 0
  1073→      ? Math.round(playbooksWithRuns.reduce((a, p) => a + p.avgDuration, 0) / playbooksWithRuns.length)
