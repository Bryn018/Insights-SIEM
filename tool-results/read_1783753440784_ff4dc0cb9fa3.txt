   161→
   162→export function AlertsView() {
   163→  const {
   164→    alertFilters,
   165→    setAlertFilters,
   166→    setAlertFilter,
   167→    resetAlertFilters,
   168→    toggleAlertSelection,
   169→    selectAllAlerts,
   170→    clearAlertSelection,
   171→    expandedAlert,
   172→    setExpandedAlert,
   173→  } = useSIEMStore()
   174→
   175→  const [alerts, setAlerts] = useState<Alert[]>([])
   176→  const [total, setTotal] = useState(0)
   177→  const [loading, setLoading] = useState(true)
   178→  const [detailData, setDetailData] = useState<Alert | null>(null)
   179→  const [detailLoading, setDetailLoading] = useState(false)
   180→  const [createDialogOpen, setCreateDialogOpen] = useState(false)
   181→  const [createForm, setCreateForm] = useState<CreateAlertForm>(emptyForm)
   182→  const [creating, setCreating] = useState(false)
   183→  const [dateRange, setDateRange] = useState<string>('24h')
   184→  const [commentText, setCommentText] = useState<Record<string, string>>({})
   185→  const [rawLogExpanded, setRawLogExpanded] = useState<Record<string, boolean>>({})
   186→  const [drawerAlertId, setDrawerAlertId] = useState<string | null>(null)
   187→
   188→  const totalPages = Math.ceil(total / alertFilters.pageSize)
   189→
   190→  // ===== Compute active filter count =====
   191→  const activeFilterCount = useMemo(() => {
   192→    let count = 0
   193→    if (alertFilters.severity.length > 0) count++
   194→    if (alertFilters.status.length > 0) count++
   195→    if (alertFilters.category.length > 0) count++
   196→    if (alertFilters.source.length > 0) count++
   197→    if (alertFilters.search) count++
   198→    if (dateRange !== '24h') count++
   199→    return count
   200→  }, [alertFilters, dateRange])
   201→
   202→  // ===== Data Fetching =====
   203→  const fetchAlerts = useCallback(async () => {
   204→    setLoading(true)
   205→    try {
   206→      const params = new URLSearchParams()
   207→      params.set('page', String(alertFilters.page))
   208→      params.set('pageSize', String(alertFilters.pageSize))
   209→      params.set('sortBy', alertFilters.sortBy)
   210→      params.set('sortOrder', alertFilters.sortOrder)
   211→      if (alertFilters.severity.length)
   212→        params.set('severity', alertFilters.severity.join(','))
   213→      if (alertFilters.status.length)
   214→        params.set('status', alertFilters.status.join(','))
   215→      if (alertFilters.category.length)
   216→        params.set('category', alertFilters.category.join(','))
   217→      if (alertFilters.source.length)
   218→        params.set('source', alertFilters.source.join(','))
   219→      if (alertFilters.search) params.set('search', alertFilters.search)
   220→      if (alertFilters.dateFrom)
   221→        params.set('dateFrom', alertFilters.dateFrom)
   222→      if (alertFilters.dateTo) params.set('dateTo', alertFilters.dateTo)
   223→
   224→      const res = await fetch(`/api/alerts?${params}`)
   225→      if (res.ok) {
   226→        const json: PaginatedResponse<Alert> = await res.json()
   227→        setAlerts(json.data || [])
   228→        setTotal(json.pagination?.total || 0)
   229→      }
   230→    } catch {
   231→      // ignore
   232→    } finally {
   233→      setLoading(false)
   234→    }
   235→  }, [
   236→    alertFilters.page,
   237→    alertFilters.pageSize,
   238→    alertFilters.sortBy,
   239→    alertFilters.sortOrder,
   240→    alertFilters.severity,
   241→    alertFilters.status,
   242→    alertFilters.category,
   243→    alertFilters.source,
   244→    alertFilters.search,
   245→    alertFilters.dateFrom,
   246→    alertFilters.dateTo,
   247→  ])
   248→
   249→  useEffect(() => {
   250→    fetchAlerts()
   251→  }, [fetchAlerts])
   252→
   253→  // ===== Fetch Alert Detail =====
   254→  const fetchDetail = useCallback(async (id: string) => {
   255→    setDetailLoading(true)
   256→    try {
   257→      const res = await fetch(`/api/alerts/${id}`)
   258→      if (res.ok) {
   259→        const json = await res.json()
   260→        setDetailData(json)
   261→      }
   262→    } catch {
   263→      // ignore
   264→    } finally {
   265→      setDetailLoading(false)
   266→    }
   267→  }, [])
   268→
   269→  // ===== Handlers =====
   270→  const handleExpand = useCallback(
   271→    (id: string) => {
   272→      if (expandedAlert === id) {
   273→        setExpandedAlert(null)
   274→      } else {
   275→        setExpandedAlert(id)
   276→        fetchDetail(id)
   277→      }
   278→    },
   279→    [expandedAlert, setExpandedAlert, fetchDetail]
   280→  )
   281→
   282→  const handleStatusChange = useCallback(
   283→    async (alertId: string, status: string) => {
   284→      try {
   285→        const res = await fetch(`/api/alerts/${alertId}`, {
   286→          method: 'PATCH',
   287→          headers: { 'Content-Type': 'application/json' },
   288→          body: JSON.stringify({ status }),
   289→        })
   290→        if (res.ok) {
   291→          toast.success(`Alert status updated to ${status}`)
   292→          fetchAlerts()
   293→          if (expandedAlert === alertId) fetchDetail(alertId)
   294→        }
   295→      } catch {
   296→        toast.error('Failed to update alert status')
   297→      }
   298→    },
   299→    [fetchAlerts, fetchDetail, expandedAlert]
   300→  )
   301→
   302→  const handleBulkAction = useCallback(
   303→    async (action: 'acknowledge' | 'escalate' | 'suppress' | 'create_incident') => {
   304→      const ids = alertFilters.selectedIds
   305→      if (ids.length === 0) return
   306→
   307→      let successCount = 0
   308→      for (const id of ids) {
   309→        try {
   310→          if (action === 'acknowledge') {
   311→            const res = await fetch(`/api/alerts/${id}`, {
   312→              method: 'PATCH',
   313→              headers: { 'Content-Type': 'application/json' },
   314→              body: JSON.stringify({ status: 'acknowledged' }),
   315→            })
   316→            if (res.ok) successCount++
   317→          } else if (action === 'escalate') {
   318→            const res = await fetch(`/api/alerts/${id}`, {
   319→              method: 'PATCH',
   320→              headers: { 'Content-Type': 'application/json' },
   321→              body: JSON.stringify({ status: 'escalated' }),
   322→            })
   323→            if (res.ok) successCount++
   324→          } else if (action === 'suppress') {
   325→            const res = await fetch(`/api/alerts/${id}`, {
   326→              method: 'PATCH',
   327→              headers: { 'Content-Type': 'application/json' },
   328→              body: JSON.stringify({ status: 'suppressed' }),
   329→            })
   330→            if (res.ok) successCount++
   331→          } else if (action === 'create_incident') {
   332→            // Open create incident dialog from store
   333→            useSIEMStore.getState().setCreateIncidentOpen(true)
   334→            return
   335→          }
   336→        } catch {
   337→          // ignore individual failures
   338→        }
   339→      }
   340→      if (action !== 'create_incident') {
   341→        toast.success(`${successCount} alerts ${action}d`)
   342→        clearAlertSelection()
   343→        fetchAlerts()
   344→      }
   345→    },
   346→    [alertFilters.selectedIds, clearAlertSelection, fetchAlerts]
   347→  )
   348→
   349→  const toggleSeverityFilter = useCallback(
   350→    (sev: Severity) => {
   351→      const current = alertFilters.severity
   352→      setAlertFilters({
   353→        severity: current.includes(sev)
   354→          ? current.filter((s) => s !== sev)
   355→          : [...current, sev],
   356→        page: 1,
   357→      })
   358→    },
   359→    [alertFilters.severity, setAlertFilters]
   360→  )
   361→
   362→  const toggleStatusFilter = useCallback(
   363→    (status: AlertStatus) => {
   364→      const current = alertFilters.status
   365→      setAlertFilters({
   366→        status: current.includes(status)
   367→          ? current.filter((s) => s !== status)
   368→          : [...current, status],
   369→        page: 1,
   370→      })
   371→    },
   372→    [alertFilters.status, setAlertFilters]
   373→  )
   374→
   375→  const toggleCategoryFilter = useCallback(
   376→    (category: string) => {
   377→      const current = alertFilters.category
   378→      setAlertFilters({
   379→        category: current.includes(category)
   380→          ? current.filter((c) => c !== category)
   381→          : [...current, category],
   382→        page: 1,
   383→      })
   384→    },
   385→    [alertFilters.category, setAlertFilters]
   386→  )
   387→
   388→  const toggleSourceFilter = useCallback(
   389→    (source: string) => {
   390→      const current = alertFilters.source
   391→      setAlertFilters({
   392→        source: current.includes(source)
   393→          ? current.filter((s) => s !== source)
   394→          : [...current, source],
   395→        page: 1,
   396→      })
   397→    },
   398→    [alertFilters.source, setAlertFilters]
   399→  )
   400→
   401→  const handleDateRange = useCallback(
   402→    (range: string) => {
   403→      setDateRange(range)
   404→      const option = DATE_RANGE_OPTIONS.find((o) => o.value === range)
   405→      if (option) {
   406→        const now = new Date()
   407→        const from = new Date(now.getTime() - option.hours * 60 * 60 * 1000)
   408→        setAlertFilters({
   409→          dateFrom: from.toISOString(),
   410→          dateTo: now.toISOString(),
   411→          page: 1,
   412→        })
   413→      }
   414→    },
   415→    [setAlertFilters]
   416→  )
   417→
   418→  const handleCreateAlert = useCallback(async () => {
   419→    if (!createForm.title || !createForm.description || !createForm.source) {
   420→      toast.error('Please fill in all required fields')
   421→      return
   422→    }
   423→    setCreating(true)
   424→    try {
   425→      const res = await fetch('/api/alerts', {
   426→        method: 'POST',
   427→        headers: { 'Content-Type': 'application/json' },
   428→        body: JSON.stringify({
   429→          title: createForm.title,
   430→          description: createForm.description,
   431→          severity: createForm.severity,
   432→          category: createForm.category,
   433→          source: createForm.source,
   434→          sourceIp: createForm.sourceIp || undefined,
   435→          destIp: createForm.destIp || undefined,
   436→          protocol: createForm.protocol || undefined,
   437→        }),
   438→      })
   439→      if (res.ok) {
   440→        toast.success('Alert created successfully')
   441→        setCreateDialogOpen(false)
   442→        setCreateForm(emptyForm)
   443→        fetchAlerts()
   444→      } else {
   445→        toast.error('Failed to create alert')
   446→      }
   447→    } catch {
   448→      toast.error('Failed to create alert')
   449→    } finally {
   450→      setCreating(false)
   451→    }
   452→  }, [createForm, fetchAlerts])
   453→
   454→  const handleAddComment = useCallback(
   455→    async (alertId: string) => {
   456→      const text = commentText[alertId]?.trim()
   457→      if (!text) return
   458→
   459→      try {
   460→        const res = await fetch(`/api/alerts/${alertId}`, {
   461→          method: 'PATCH',
   462→          headers: { 'Content-Type': 'application/json' },
   463→          body: JSON.stringify({ comment: text }),
   464→        })
   465→        if (res.ok) {
   466→          toast.success('Comment added')
   467→          setCommentText((prev) => ({ ...prev, [alertId]: '' }))
   468→          fetchDetail(alertId)
   469→        }
   470→      } catch {
   471→        toast.error('Failed to add comment')
   472→      }
   473→    },
   474→    [commentText, fetchDetail]
   475→  )
   476→
   477→  // Select all on current page
   478→  const allSelected =
   479→    alerts.length > 0 && alerts.every((a) => alertFilters.selectedIds.includes(a.id))
   480→  const someSelected =
   481→    alerts.some((a) => alertFilters.selectedIds.includes(a.id)) && !allSelected
   482→
   483→  // ===== Render =====
   484→  return (
   485→    <motion.div
   486→      initial={{ opacity: 0 }}
   487→      animate={{ opacity: 1 }}
   488→      className="flex flex-col gap-4 p-4 md:p-6"
   489→    >
   490→      {/* Header Row */}
   491→      <div className="flex items-center justify-between">
   492→        <div className="flex items-center gap-2">
   493→          <h2 className="text-lg font-semibold">Alerts</h2>
   494→          {activeFilterCount > 0 && (
   495→            <Badge
   496→              variant="secondary"
   497→              className="bg-emerald-500/15 text-emerald-400 border-emerald-500/30 text-[10px] h-5 gap-1"
   498→            >
   499→              <Filter className="h-2.5 w-2.5" />
   500→              {activeFilterCount} active
   501→            </Badge>
   502→          )}
   503→        </div>
   504→        <div className="flex items-center gap-2">
   505→          <ExportButton
   506→            filename="siem-alerts"
   507→            fetchData={async () => {
   508→              // Reuse the same query params as the current view, but request
   509→              // a large page size to get all matching rows in one shot.
   510→              const params = new URLSearchParams()
   511→              params.set('page', '1')
   512→              params.set('pageSize', '10000')
   513→              params.set('sortBy', alertFilters.sortBy)
   514→              params.set('sortOrder', alertFilters.sortOrder)
   515→              if (alertFilters.severity.length)
   516→                params.set('severity', alertFilters.severity.join(','))
   517→              if (alertFilters.status.length)
   518→                params.set('status', alertFilters.status.join(','))
   519→              if (alertFilters.category.length)
   520→                params.set('category', alertFilters.category.join(','))
   521→              if (alertFilters.source.length)
   522→                params.set('source', alertFilters.source.join(','))
   523→              if (alertFilters.search) params.set('search', alertFilters.search)
   524→              const res = await fetch(`/api/alerts?${params.toString()}`)
   525→              if (!res.ok) throw new Error('Failed to fetch alerts for export')
   526→              const json = await res.json()
   527→              const rows = (json.data ?? json.alerts ?? []) as Record<string, unknown>[]
   528→              return rows.map((r) => ({
   529→                id: r.id,
   530→                title: r.title,
   531→                severity: r.severity,
   532→                status: r.status,
   533→                category: r.category,
   534→                source: r.source,
   535→                sourceIp: r.sourceIp,
   536→                destIp: r.destIp,
   537→                hostname: r.hostname,
   538→                createdAt: r.createdAt,
   539→                updatedAt: r.updatedAt,
   540→              }))
   541→            }}
   542→          />
   543→          <Button
   544→            size="sm"
   545→            className="gap-1.5 bg-emerald-600 hover:bg-emerald-700 text-white"
   546→            onClick={() => setCreateDialogOpen(true)}
   547→          >
   548→            <Plus className="h-3.5 w-3.5" />
   549→            Create Alert
   550→          </Button>
   551→        </div>
   552→      </div>
   553→
   554→      {/* Filter Bar */}
   555→      <Card className="border-border bg-card">
   556→        <CardContent className="p-3">
   557→          <div className="flex flex-wrap items-center gap-2">
   558→            {/* Severity Multi-Select Buttons */}
   559→            {SEVERITY_OPTIONS.map((opt) => {
   560→              const isSelected = alertFilters.severity.includes(opt.value)
   561→              return (
   562→                <button
   563→                  key={opt.value}
   564→                  onClick={() => toggleSeverityFilter(opt.value)}
   565→                  className={cn(
   566→                    'inline-flex items-center gap-1.5 rounded-md border px-2.5 py-1 text-xs font-medium transition-all',
   567→                    isSelected
   568→                      ? severityButtonColors[opt.value].active
   569→                      : 'border-border text-muted-foreground hover:text-foreground hover:border-foreground/20'
   570→                  )}
   571→                >
   572→                  <span
   573→                    className={cn(
   574→                      'h-2 w-2 rounded-full',
   575→                      isSelected ? opt.color : 'bg-current opacity-30'
   576→                    )}
   577→                  />
   578→                  {opt.label}
   579→                </button>
   580→              )
   581→            })}
   582→
   583→            <Separator orientation="vertical" className="h-6 mx-1" />
   584→
   585→            {/* Status Dropdown with Checkboxes */}
   586→            <Popover>
   587→              <PopoverTrigger asChild>
   588→                <Button
   589→                  variant="outline"
   590→                  size="sm"
   591→                  className="h-8 gap-1.5 text-xs"
   592→                >
