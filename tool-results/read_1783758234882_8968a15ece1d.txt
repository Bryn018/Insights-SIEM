   111→function CircularScoreRing({
   112→  score,
   113→  size = 120,
   114→  strokeWidth = 10,
   115→  label,
   116→  showValue = true,
   117→  className,
   118→}: {
   119→  score: number
   120→  size?: number
   121→  strokeWidth?: number
   122→  label?: string
   123→  showValue?: boolean
   124→  className?: string
   125→}) {
   126→  const radius = (size - strokeWidth) / 2
   127→  const circumference = 2 * Math.PI * radius
   128→  const offset = circumference - (Math.min(score, 100) / 100) * circumference
   129→
   130→  const getColor = () => {
   131→    if (score >= 80) return '#10b981'
   132→    if (score >= 60) return '#eab308'
   133→    if (score >= 40) return '#f97316'
   134→    return '#ef4444'
   135→  }
   136→
   137→  const getGlowColor = () => {
   138→    if (score >= 80) return 'rgba(16, 185, 129, 0.2)'
   139→    if (score >= 60) return 'rgba(234, 179, 8, 0.2)'
   140→    if (score >= 40) return 'rgba(249, 115, 22, 0.2)'
   141→    return 'rgba(239, 68, 68, 0.2)'
   142→  }
   143→
   144→  const color = getColor()
   145→
   146→  return (
   147→    <div className={cn('relative inline-flex flex-col items-center justify-center', className)}>
   148→      <svg width={size} height={size} className="-rotate-90">
   149→        <circle
   150→          cx={size / 2}
   151→          cy={size / 2}
   152→          r={radius + strokeWidth / 2}
   153→          fill="none"
   154→          stroke={getGlowColor()}
   155→          strokeWidth={strokeWidth * 2}
   156→          className="blur-sm"
   157→        />
   158→        <circle
   159→          cx={size / 2}
   160→          cy={size / 2}
   161→          r={radius}
   162→          fill="none"
   163→          stroke="currentColor"
   164→          strokeWidth={strokeWidth}
   165→          className="text-muted/20"
   166→        />
   167→        <motion.circle
   168→          cx={size / 2}
   169→          cy={size / 2}
   170→          r={radius}
   171→          fill="none"
   172→          stroke={color}
   173→          strokeWidth={strokeWidth}
   174→          strokeDasharray={circumference}
   175→          strokeDashoffset={offset}
   176→          strokeLinecap="round"
   177→          initial={{ strokeDashoffset: circumference }}
   178→          animate={{ strokeDashoffset: offset }}
   179→          transition={{ duration: 1.2, ease: 'easeOut' }}
   180→        />
   181→      </svg>
   182→      {showValue && (
   183→        <div className="absolute flex flex-col items-center justify-center">
   184→          <motion.span
   185→            className="font-bold"
   186→            style={{ color, fontSize: size * 0.22 }}
   187→            initial={{ opacity: 0 }}
   188→            animate={{ opacity: 1 }}
   189→            transition={{ delay: 0.5 }}
   190→          >
   191→            {score}%
   192→          </motion.span>
   193→          {label && (
   194→            <span className="text-[10px] text-muted-foreground mt-0.5">{label}</span>
   195→          )}
   196→        </div>
   197→      )}
   198→    </div>
   199→  )
   200→}

   304→// ===== Safe relative-time formatter (returns short "5m ago" style) =====
   305→
   306→function formatDistanceToNowSafe(dateStr: string): string {
   307→  try {
   308→    const then = new Date(dateStr).getTime()
   309→    if (Number.isNaN(then)) return ''
   310→    const diffMs = Date.now() - then
   311→    if (diffMs < 0) return 'just now'
   312→    const sec = Math.floor(diffMs / 1000)
   313→    if (sec < 60) return 'just now'
   314→    const min = Math.floor(sec / 60)
   315→    if (min < 60) return `${min}m ago`
   316→    const hr = Math.floor(min / 60)
   317→    if (hr < 24) return `${hr}h ago`
   318→    const day = Math.floor(hr / 24)
   319→    if (day < 30) return `${day}d ago`
   320→    const mo = Math.floor(day / 30)
   321→    if (mo < 12) return `${mo}mo ago`
   322→    return `${Math.floor(mo / 12)}y ago`
   323→  } catch {
   324→    return ''
   325→  }
   326→}

   344→// ===== 30-Day Compliance Trend Data =====
   345→
   346→function generate30DayTrendData(frameworks: FrameworkSummary[]) {
   347→  const days = 30
   348→  const today = new Date()
   349→  return Array.from({ length: days }, (_, i) => {
   350→    const date = subDays(today, days - 1 - i)
   351→    const entry: Record<string, string | number> = {
   352→      date: format(date, 'MMM d'),
   353→      fullDate: format(date, 'yyyy-MM-dd'),
   354→    }
   355→    frameworks.forEach((fw) => {
   356→      const baseScore = fw.complianceScore
   357→      const dayVariance = Math.sin(i * 0.3) * 4 + (Math.random() - 0.5) * 6
   358→      const trendAdjust = (i / days) * 5
   359→      entry[fw.name] = Math.min(100, Math.max(0, Math.round(baseScore - 5 + dayVariance + trendAdjust)))
   360→    })
   361→    return entry
   362→  })
   363→}

   365→// ===== 6-Month Trend Data (for line chart fallback) =====

   367→function generate6MonthTrendData(frameworks: FrameworkSummary[]) {
   368→  const months = ['Oct', 'Nov', 'Dec', 'Jan', 'Feb', 'Mar']
   369→  return months.map((month, i) => {
   370→    const entry: Record<string, string | number> = { month }
   371→    frameworks.forEach((fw) => {
   372→      const baseScore = fw.complianceScore
   373→      const variance = Math.round((Math.random() - 0.4) * 12)
   374→      const trendAdjust = i * 1.5
   375→      entry[fw.name] = Math.min(100, Math.max(0, baseScore + variance + trendAdjust))
   376→    })
   377→    return entry
   378→  })

   381→// ===== Radar Chart Data Generator =====
   382→
   383→function generateRadarData(frameworks: FrameworkSummary[]) {
   384→  const dimensions = [
   385→    { key: 'compliance', label: 'Compliance' },
   386→    { key: 'coverage', label: 'Coverage' },
   387→    { key: 'evidence', label: 'Evidence' },
   388→    { key: 'remediation', label: 'Remediation' },
   389→    { key: 'monitoring', label: 'Monitoring' },
   390→    { key: 'governance', label: 'Governance' },
   391→  ]
   392→
   393→  return dimensions.map((dim) => {
   394→    const entry: Record<string, string | number> = { dimension: dim.label }
   395→    frameworks.forEach((fw) => {
   396→      const base = fw.complianceScore
   397→      let value: number
   398→      switch (dim.key) {
   399→        case 'compliance':
   400→          value = base
   401→          break
   402→        case 'coverage': {
   403→          const assessed = (fw.statusCounts.compliant || 0) + (fw.statusCounts.non_compliant || 0) + (fw.statusCounts.partially_compliant || 0)
   404→          value = fw.totalControls > 0 ? Math.round((assessed / fw.totalControls) * 100) : 0
   405→          break
   406→        }
   407→        case 'evidence':
   408→          value = Math.min(100, base + Math.round((Math.random() - 0.3) * 15))
   409→          break
   410→        case 'remediation': {
   411→          const nonCompliant = fw.statusCounts.non_compliant || 0
   412→          value = nonCompliant > 0 ? Math.round(Math.max(20, 100 - nonCompliant * 10)) : 95
   413→          break
   414→        }
   415→        case 'monitoring':
   416→          value = Math.min(100, base + Math.round((Math.random() - 0.4) * 10))
   417→          break
   418→        case 'governance':
   419→          value = Math.min(100, base + Math.round((Math.random() - 0.5) * 12))
   420→          break
   421→        default:
   422→          value = base
   423→      }
   424→      entry[fw.name] = Math.max(0, Math.min(100, value))
   425→    })
   426→    return entry
   427→  })
   428→}

   466→// ===== Control Row with Inline Status Select =====
   467→
   468→function ControlRow({
   469→  control,
   470→  frameworkId,
   471→  onStatusChange,
   472→}: {
   473→  control: ControlData
   474→  frameworkId: string
   475→  onStatusChange: (controlId: string, newStatus: ControlStatus) => void
   476→}) {
   477→  const [statusUpdating, setStatusUpdating] = useState(false)
   478→  const [expanded, setExpanded] = useState(false)
   479→
   480→  const assessmentProgress = useMemo(() => {
   481→    if (control.status === 'compliant' || control.status === 'non_compliant') return 100
   482→    if (control.status === 'partially_compliant') return 60
   483→    if (control.status === 'not_applicable') return 100
   484→    return 0
   485→  }, [control.status])
   486→
   487→  const handleStatusChange = async (newStatus: string) => {
   488→    if (newStatus === control.status) return
   489→    setStatusUpdating(true)
   490→    try {
   491→      const res = await fetch(`/api/compliance/${frameworkId}/controls`, {
   492→        method: 'PATCH',
   493→        headers: { 'Content-Type': 'application/json' },
   494→        body: JSON.stringify({
   495→          controlId: control.id,
   496→          status: newStatus,
   497→        }),
   498→      })
   499→      if (res.ok) {
   500→        toast.success(`Control ${control.controlId} updated to ${newStatus.replace(/_/g, ' ')}`)
   501→        onStatusChange(control.id, newStatus as ControlStatus)
   502→      } else {
   503→        toast.error('Failed to update control status')
   504→      }
   505→    } catch {
   506→      toast.error('Failed to update control status')
   507→    } finally {
   508→      setStatusUpdating(false)
   509→    }
   510→  }
   511→
   512→  return (
   513→    <motion.div
   514→      layout
   515→      initial={{ opacity: 0 }}
   516→      animate={{ opacity: 1 }}
   517→      className={cn(
   518→        'rounded-lg border transition-colors',
   519→        control.status === 'non_compliant'
   520→          ? 'border-red-500/20 bg-red-500/5 border-l-2 border-l-red-500/50'
   521→          : control.status === 'partially_compliant'
   522→            ? 'border-amber-500/20 bg-amber-500/5'
   523→            : 'border-border/50 bg-muted/10'
   524→      )}
   525→    >
   526→      <div
   527→        className="flex items-center gap-3 px-3 py-2.5 cursor-pointer hover:bg-accent/20 rounded-lg"
   528→        onClick={() => setExpanded(!expanded)}
   529→      >
   530→        {/* Expand chevron */}
   531→        <motion.div
   532→          animate={{ rotate: expanded ? 90 : 0 }}
   533→          transition={{ duration: 0.15 }}
   534→          className="shrink-0"
   535→        >
   536→          <ChevronRight className="h-3.5 w-3.5 text-muted-foreground" />
   537→        </motion.div>
   538→
   539→        {/* Status icon */}
   540→        <StatusIcon status={control.status} />
   541→
   542→        {/* Control ID */}
   543→        <span className="font-mono text-xs text-muted-foreground shrink-0 w-20">
   544→          {control.controlId}
   545→        </span>
   546→
   547→        {/* Title */}
   548→        <span className="text-xs flex-1 truncate">{control.title}</span>
   549→
   550→        {/* Category badge (hidden on small) */}
   551→        {control.category && (
   552→          <Badge variant="outline" className="text-[9px] px-1.5 py-0 hidden md:inline-flex shrink-0">
   553→            {control.category}
   554→          </Badge>
   555→        )}
   556→
   557→        {/* Mini assessment progress */}
   558→        <div className="hidden sm:flex items-center gap-1.5 shrink-0 w-20">
   559→          <Progress
   560→            value={assessmentProgress}
   561→            className={cn(
   562→              'h-1 w-14',
   563→              control.status === 'compliant'
   564→                ? '[&>div]:bg-emerald-500'
   565→                : control.status === 'non_compliant'
   566→                  ? '[&>div]:bg-red-500'
   567→                  : control.status === 'partially_compliant'
   568→                    ? '[&>div]:bg-amber-500'
   569→                    : '[&>div]:bg-zinc-500'
   570→            )}
   571→          />
   572→          <span className="text-[9px] text-muted-foreground tabular-nums">{assessmentProgress}%</span>
   573→        </div>
   574→
   575→        {/* Status select */}
   576→        <div onClick={(e) => e.stopPropagation()} className="shrink-0">
   577→          <Select
   578→            value={control.status}
   579→            onValueChange={handleStatusChange}
   580→            disabled={statusUpdating}
   581→          >
   582→            <SelectTrigger className={cn(
   583→              'h-6 w-[130px] text-[10px] border-0 bg-transparent',
   584→              'hover:bg-accent/30 focus:ring-0 focus:ring-offset-0',
   585→              statusUpdating && 'opacity-50'
   586→            )}>
   587→              <SelectValue />
   588→            </SelectTrigger>
   589→            <SelectContent className="text-xs">
   590→              <SelectItem value="compliant">
   591→                <span className="flex items-center gap-1.5">
   592→                  <CheckCircle2 className="h-3 w-3 text-emerald-400" /> Compliant
   593→                </span>
   594→              </SelectItem>
   595→              <SelectItem value="non_compliant">
   596→                <span className="flex items-center gap-1.5">
   597→                  <XCircle className="h-3 w-3 text-red-400" /> Non-Compliant
   598→                </span>
   599→              </SelectItem>
   600→              <SelectItem value="partially_compliant">
   601→                <span className="flex items-center gap-1.5">
   602→                  <AlertTriangle className="h-3 w-3 text-amber-400" /> Partial
