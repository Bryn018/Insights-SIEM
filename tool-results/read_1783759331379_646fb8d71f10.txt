   133→function generateMockResults(query: string, count: number): ThreatHuntResult[] {
   134→  const now = Date.now()
   135→  const severities: Severity[] = ['critical', 'high', 'medium', 'low', 'informational']
   136→  // Weight toward higher severity for more interesting results
   137→  const weights = [0.1, 0.25, 0.35, 0.2, 0.1]
   138→
   139→  return Array.from({ length: count }, (_, i) => {
   140→    let severity: Severity = 'informational'
   141→    const rand = Math.random()
   142→    let cumulative = 0
   143→    for (let s = 0; s < severities.length; s++) {
   144→      cumulative += weights[s]
   145→      if (rand <= cumulative) {
   146→        severity = severities[s]
   147→        break
   148→      }
   149→    }
   150→
   151→    const timeOffset = Math.random() * 86400000 // up to 24h
   152→    return {
   153→      id: `hunt-${Date.now()}-${i}`,
   154→      timestamp: new Date(now - timeOffset).toISOString(),
   155→      source: MOCK_SOURCES[Math.floor(Math.random() * MOCK_SOURCES.length)],
   156→      severity,
   157→      message: MOCK_MESSAGES[Math.floor(Math.random() * MOCK_MESSAGES.length)],
   158→      host: MOCK_HOSTS[Math.floor(Math.random() * MOCK_HOSTS.length)],
   159→      user: MOCK_USERS[Math.floor(Math.random() * MOCK_USERS.length)],
   160→    }
   161→  }).sort((a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime())
   162→}

   166→interface IOC {
   167→  type: 'IP' | 'Hash' | 'Domain' | 'URL'
   168→  value: string
   169→}
   170→
   171→function extractIOCs(results: ThreatHuntResult[]): IOC[] {
   172→  const seen = new Set<string>()
   173→
   174→  // Extract IPs from host field (simulated)
   175→  const ipPattern = /\b(?:\d{1,3}\.){3}\d{1,3}\b/g
   176→  // Extract hashes from messages (simulated MD5/SHA patterns)
   177→  const hashPattern = /\b[a-fA-F0-9]{32,64}\b/g
   178→  // Extract domains
   179→  const domainPattern = /\b(?:[a-z0-9](?:[a-z0-9-]{0,61}[a-z0-9])?\.)+[a-z]{2,}\b/g
   180→  // Extract URLs
   181→  const urlPattern = /https?:\/\/[^\s<>"]+/g
   182→
   183→  const allText = results.map(r => `${r.message} ${r.host} ${r.user} ${r.source}`).join(' ')
   184→
   185→  // Add some simulated IOCs based on the results for demo purposes
   186→  const resultIPs = [
   187→    '10.0.1.45', '10.0.2.100', '192.168.1.55', '172.16.0.23',
   188→    '203.0.113.50', '198.51.100.12', '10.0.0.99', '192.168.100.5',
   189→  ]
   190→  const resultHashes = [
   191→    'a3f2c8e9b1d4f6a7c8e9b1d4f6a7c8e9',
   192→    '5e884898da28047151d0e56f8dc6292773603d0d6aabbdd62a11ef721d1542d8',
   193→    'e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855',
   194→  ]
   195→  const resultDomains = [
   196→    'malware-c2.evil.com', 'update.suspicious-domain.net',
   197→    'cdn.malware-distro.org', 'beacon.adversary.io',
   198→  ]
   199→  const resultURLs = [
   200→    'http://10.0.1.45:4444/shell', 'https://malware-c2.evil.com/api/beacon',
   201→    'http://203.0.113.50/payload.exe',
   202→  ]
   203→
   204→  // Pick a subset based on result count
   205→  const ipCount = Math.min(resultIPs.length, Math.ceil(results.length / 8))
   206→  const hashCount = Math.min(resultHashes.length, Math.ceil(results.length / 15))
   207→  const domainCount = Math.min(resultDomains.length, Math.ceil(results.length / 12))
   208→  const urlCount = Math.min(resultURLs.length, Math.ceil(results.length / 20))
   209→
   210→  const addIOC = (type: IOC['type'], value: string) => {
   211→    const key = `${type}:${value}`
   212→    if (!seen.has(key)) {
   213→      seen.add(key)
   214→    }
   215→  }
   216→
   217→  resultIPs.slice(0, ipCount).forEach(ip => addIOC('IP', ip))
   218→  resultHashes.slice(0, hashCount).forEach(h => addIOC('Hash', h))
   219→  resultDomains.slice(0, domainCount).forEach(d => addIOC('Domain', d))
   220→  resultURLs.slice(0, urlCount).forEach(u => addIOC('URL', u))
   221→
   222→  // Also extract from the actual text
   223→  let match: RegExpExecArray | null
   224→  ipPattern.lastIndex = 0
   225→  while ((match = ipPattern.exec(allText)) !== null) addIOC('IP', match[0])
   226→  hashPattern.lastIndex = 0
   227→  while ((match = hashPattern.exec(allText)) !== null) addIOC('Hash', match[0])
   228→  domainPattern.lastIndex = 0
   229→  while ((match = domainPattern.exec(allText)) !== null) addIOC('Domain', match[0])
   230→  urlPattern.lastIndex = 0
   231→  while ((match = urlPattern.exec(allText)) !== null) addIOC('URL', match[0])
   232→
   233→  return Array.from(seen).map(key => {
   234→    const [type, ...rest] = key.split(':')
   235→    return { type: type as IOC['type'], value: rest.join(':') }
   236→  })
   237→}

   377→// ===== Hunt Execution =====

   379→async function executeHunt(
   380→  state: { query: string; timeRange: string; dataSource: string; huntHistory: ThreatHuntResult[] },
   381→  setter: (partial: Record<string, unknown>) => void
   382→) {
   383→  if (!state.query.trim()) return
   384→
   385→  setter({ isHunting: true, results: [] })
   386→
   387→  // Simulate hunt execution delay
   388→  await new Promise(resolve => setTimeout(resolve, 1500 + Math.random() * 2000))
   389→
   390→  const resultCount = 20 + Math.floor(Math.random() * 30)
   391→  const results = generateMockResults(state.query, resultCount)
   392→
   393→  // Add to history
   394→  const historyEntry = {
   395→    query: state.query,
   396→    time: new Date().toISOString(),
   397→    resultCount: results.length,
   398→    dataSource: state.dataSource,
   399→  }
   400→
   401→  // Use the setter with proper types
   402→  const currentHistory = (state as Record<string, unknown>).huntHistory as Array<Record<string, unknown>> || []
   403→  setter({
   404→    results,
   405→    isHunting: false,
   406→    huntHistory: [historyEntry, ...currentHistory].slice(0, 10),
   407→  })
   408→
   409→  toast.success(`Hunt complete: ${results.length} matches found`, {
   410→    description: `Query: ${state.query.substring(0, 60)}${state.query.length > 60 ? '...' : ''}`,
   411→  })
   412→}

   863→// ===== Hunt Intensity Indicator =====
   864→
   865→function HuntIntensityBar({ isHunting, resultCount }: { isHunting: boolean; resultCount: number }) {
   866→  // Intensity is derived from the result count + active hunting state.
   867→  // More results = higher intensity (capped at 8 segments lit).
   868→  const intensity = isHunting
   869→    ? 8
   870→    : resultCount === 0
   871→      ? 0
   872→      : Math.min(8, Math.max(1, Math.ceil(resultCount / 8)))
   873→  const label = isHunting ? 'SCANNING' : intensity === 0 ? 'IDLE' : intensity >= 6 ? 'HIGH' : intensity >= 3 ? 'MEDIUM' : 'LOW'
   874→  const labelColor = isHunting
   875→    ? 'text-emerald-400'
   876→    : intensity >= 6
   877→      ? 'text-red-400'
   878→      : intensity >= 3
   879→        ? 'text-amber-400'
   880→        : intensity > 0
   881→          ? 'text-zinc-300'
   882→          : 'text-zinc-600'
   883→
   884→  return (
   885→    <div className="flex items-center gap-2 rounded-md border border-zinc-800 bg-zinc-950/60 px-3 py-1.5">
   886→      <span className="text-[9px] uppercase tracking-wider text-zinc-500">Hunt Intensity</span>
   887→      <div className="flex items-end gap-0.5 h-3.5" aria-hidden="true">
   888→        {Array.from({ length: 8 }).map((_, i) => {
   889→          const lit = i < intensity
   890→          const height = 30 + i * 10 // 30%, 40%, ... 100%
   891→          return (
   892→            <span
   893→              key={i}
   894→              className={cn(
   895→                'w-1 rounded-sm transition-colors',
   896→                lit
   897→                  ? intensity >= 6
   898→                    ? 'bg-red-400'
   899→                    : intensity >= 3
   900→                      ? 'bg-amber-400'
   901→                      : 'bg-emerald-400'
   902→                  : 'bg-zinc-800'
   903→              )}
   904→              style={{
   905→                height: `${height}%`,
   906→                animationDelay: `${i * 90}ms`,
   907→              }}
   908→            >
   909→              {lit && isHunting && (
   910→                <span
   911→                  className="siem-intensity-segment block h-full w-full"
   912→                  style={{ animationDelay: `${i * 90}ms` }}
   913→                />
   914→              )}
   915→            </span>
   916→          )
   917→        })}
   918→      </div>
   919→      <span className={cn('text-[10px] font-mono font-semibold', labelColor)}>{label}</span>
   920→    </div>
   921→  )
   922→}
